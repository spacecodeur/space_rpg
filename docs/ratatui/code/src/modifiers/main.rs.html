1/// A Ratatui example that demonstrates how to use modifiers.
2///
3/// It will render a grid of combinations of foreground and background colors with all
4/// modifiers applied to them.
5///
6/// This example runs with the Ratatui library code in the branch that you are currently
7/// reading. See the [`latest`] branch for the code which works with the most recent Ratatui
8/// release.
9///
10/// [`latest`]: https://github.com/ratatui/ratatui/tree/latest
11use std::{error::Error, iter::once, result};
12
13use crossterm::event;
14use itertools::Itertools;
15use ratatui::Frame;
16use ratatui::layout::{Constraint, Layout};
17use ratatui::style::{Color, Modifier, Style, Stylize};
18use ratatui::text::Line;
19use ratatui::widgets::Paragraph;
20
21type Result<T> = result::Result<T, Box<dyn Error>>;
22
23fn main() -> Result<()> {
24    color_eyre::install()?;
25    ratatui::run(|terminal| {
26        loop {
27            terminal.draw(render)?;
28            if event::read()?.is_key_press() {
29                break Ok(());
30            }
31        }
32    })
33}
34
35fn render(frame: &mut Frame) {
36    let layout = Layout::vertical([Constraint::Length(1), Constraint::Min(0)]);
37    let [text_area, main_area] = frame.area().layout(&layout);
38    frame.render_widget(
39        Paragraph::new("Note: not all terminals support all modifiers")
40            .style(Style::default().fg(Color::Red).add_modifier(Modifier::BOLD)),
41        text_area,
42    );
43    let layout = Layout::vertical([Constraint::Length(1); 50])
44        .split(main_area)
45        .iter()
46        .flat_map(|area| {
47            Layout::horizontal([Constraint::Percentage(20); 5])
48                .split(*area)
49                .to_vec()
50        })
51        .collect_vec();
52
53    let colors = [
54        Color::Black,
55        Color::DarkGray,
56        Color::Gray,
57        Color::White,
58        Color::Red,
59    ];
60    let all_modifiers = once(Modifier::empty())
61        .chain(Modifier::all().iter())
62        .collect_vec();
63    let mut index = 0;
64    for bg in &colors {
65        for fg in &colors {
66            for modifier in &all_modifiers {
67                let modifier_name = format!("{modifier:11?}");
68                let padding = (" ").repeat(12 - modifier_name.len());
69                let paragraph = Paragraph::new(Line::from(vec![
70                    modifier_name.fg(*fg).bg(*bg).add_modifier(*modifier),
71                    padding.fg(*fg).bg(*bg).add_modifier(*modifier),
72                    // This is a hack to work around a bug in VHS which is used for rendering the
73                    // examples to gifs. The bug is that the background color of a paragraph seems
74                    // to bleed into the next character.
75                    ".".black().on_black(),
76                ]));
77                frame.render_widget(paragraph, layout[index]);
78                index += 1;
79            }
80        }
81    }
82}
