1//! # [Ratatui] Async example
2//!
3//! This example demonstrates how to use Ratatui with widgets that fetch data asynchronously. It
4//! uses the `octocrab` crate to fetch a list of pull requests from the GitHub API.
5//!
6//! <https://docs.github.com/en/authentication/keeping-your-account-and-data-secure/managing-your-personal-access-tokens#creating-a-fine-grained-personal-access-token>
7//! <https://github.com/settings/tokens/new> to create a new token (select classic, and no scopes)
8//!
9//! This example does not cover message passing between threads, it only demonstrates how to manage
10//! shared state between the main thread and a background task, which acts mostly as a one-shot
11//! fetcher. For more complex scenarios, you may need to use channels or other synchronization
12//! primitives.
13//!
14//! A simple app might have multiple widgets that fetch data from different sources, and each widget
15//! would have its own background task to fetch the data. The main thread would then render the
16//! widgets with the latest data.
17//!
18//! The latest version of this example is available in the [examples] folder in the repository.
19//!
20//! Please note that the examples are designed to be run against the `main` branch of the Github
21//! repository. This means that you may not be able to compile with the latest release version on
22//! crates.io, or the one that you have installed locally.
23//!
24//! See the [examples readme] for more information on finding examples that match the version of the
25//! library you are using.
26//!
27//! [Ratatui]: https://github.com/ratatui/ratatui
28//! [examples]: https://github.com/ratatui/ratatui/blob/main/examples
29//! [examples readme]: https://github.com/ratatui/ratatui/blob/main/examples/README.md

46#[tokio::main]
47async fn main() -> Result<()> {
48    color_eyre::install()?;
49    let terminal = ratatui::init();
50    let app_result = App::default().run(terminal).await;
51    ratatui::restore();
52    app_result
53}
54
55#[derive(Debug, Default)]
56struct App {
57    should_quit: bool,
58    pull_requests: PullRequestListWidget,
59}
60
61impl App {
62    const FRAMES_PER_SECOND: f32 = 60.0;
63
64    pub async fn run(mut self, mut terminal: DefaultTerminal) -> Result<()> {
65        self.pull_requests.run();
66
67        let period = Duration::from_secs_f32(1.0 / Self::FRAMES_PER_SECOND);
68        let mut interval = tokio::time::interval(period);
69        let mut events = EventStream::new();
70
71        while !self.should_quit {
72            tokio::select! {
73                _ = interval.tick() => { terminal.draw(|frame| self.render(frame))?; },
74                Some(Ok(event)) = events.next() => self.handle_event(&event),
75            }
76        }
77        Ok(())
78    }
79
80    fn render(&self, frame: &mut Frame) {
81        let layout = Layout::vertical([Constraint::Length(1), Constraint::Fill(1)]);
82        let [title_area, body_area] = frame.area().layout(&layout);
83        let title = Line::from("Ratatui async example").centered().bold();
84        frame.render_widget(title, title_area);
85        frame.render_widget(&self.pull_requests, body_area);
86    }
87
88    fn handle_event(&mut self, event: &Event) {
89        if let Some(key) = event.as_key_press_event() {
90            match key.code {
91                KeyCode::Char('q') | KeyCode::Esc => self.should_quit = true,
92                KeyCode::Char('j') | KeyCode::Down => self.pull_requests.scroll_down(),
93                KeyCode::Char('k') | KeyCode::Up => self.pull_requests.scroll_up(),
94                _ => {}
95            }
96        }
97    }
98}

100/// A widget that displays a list of pull requests.
101///
102/// This is an async widget that fetches the list of pull requests from the GitHub API. It contains
103/// an inner `Arc<RwLock<PullRequestListState>>` that holds the state of the widget. Cloning the
104/// widget will clone the Arc, so you can pass it around to other threads, and this is used to spawn
105/// a background task to fetch the pull requests.
106#[derive(Debug, Clone, Default)]
107struct PullRequestListWidget {
108    state: Arc<RwLock<PullRequestListState>>,
109}
110
111#[derive(Debug, Default)]
112struct PullRequestListState {
113    pull_requests: Vec<PullRequest>,
114    loading_state: LoadingState,
115    table_state: TableState,
116}
117
118#[derive(Debug, Clone)]
119struct PullRequest {
120    id: String,
121    title: String,
122    url: String,
123}
124
125#[derive(Debug, Clone, Default, PartialEq, Eq)]
126enum LoadingState {
127    #[default]
128    Idle,
129    Loading,
130    Loaded,
131    Error(String),
132}
133
134impl PullRequestListWidget {
135    /// Start fetching the pull requests in the background.
136    ///
137    /// This method spawns a background task that fetches the pull requests from the GitHub API.
138    /// The result of the fetch is then passed to the `on_load` or `on_err` methods.
139    fn run(&self) {
140        let this = self.clone(); // clone the widget to pass to the background task
141        tokio::spawn(this.fetch_pulls());
142    }
143
144    async fn fetch_pulls(self) {
145        // this runs once, but you could also run this in a loop, using a channel that accepts
146        // messages to refresh on demand, or with an interval timer to refresh every N seconds
147        self.set_loading_state(LoadingState::Loading);
148        match octocrab::instance()
149            .pulls("ratatui", "ratatui")
150            .list()
151            .sort(Sort::Updated)
152            .direction(Direction::Descending)
153            .send()
154            .await
155        {
156            Ok(page) => self.on_load(&page),
157            Err(err) => self.on_err(&err),
158        }
159    }
160    fn on_load(&self, page: &Page<OctoPullRequest>) {
161        let prs = page.items.iter().map(Into::into);
162        let mut state = self.state.write().unwrap();
163        state.loading_state = LoadingState::Loaded;
164        state.pull_requests.extend(prs);
165        if !state.pull_requests.is_empty() {
166            state.table_state.select(Some(0));
167        }
168    }
169
170    fn on_err(&self, err: &octocrab::Error) {
171        self.set_loading_state(LoadingState::Error(err.to_string()));
172    }
173
174    fn set_loading_state(&self, state: LoadingState) {
175        self.state.write().unwrap().loading_state = state;
176    }
177
178    fn scroll_down(&self) {
179        self.state.write().unwrap().table_state.scroll_down_by(1);
180    }
181
182    fn scroll_up(&self) {
183        self.state.write().unwrap().table_state.scroll_up_by(1);
184    }
185}

187type OctoPullRequest = octocrab::models::pulls::PullRequest;
188
189impl From<&OctoPullRequest> for PullRequest {
190    fn from(pr: &OctoPullRequest) -> Self {
191        Self {
192            id: pr.number.to_string(),
193            title: pr.title.as_ref().unwrap().to_string(),
194            url: pr
195                .html_url
196                .as_ref()
197                .map(ToString::to_string)
198                .unwrap_or_default(),
199        }
200    }
201}
202
203impl Widget for &PullRequestListWidget {
204    fn render(self, area: Rect, buf: &mut Buffer) {
205        let mut state = self.state.write().unwrap();
206
207        // a block with a right aligned title with the loading state on the right
208        let loading_state = Line::from(format!("{:?}", state.loading_state)).right_aligned();
209        let block = Block::bordered()
210            .title("Pull Requests")
211            .title(loading_state)
212            .title_bottom("j/k to scroll, q to quit");
213
214        // a table with the list of pull requests
215        let rows = state.pull_requests.iter();
216        let widths = [
217            Constraint::Length(5),
218            Constraint::Fill(1),
219            Constraint::Max(49),
220        ];
221        let table = Table::new(rows, widths)
222            .block(block)
223            .highlight_spacing(HighlightSpacing::Always)
224            .highlight_symbol(">>")
225            .row_highlight_style(Style::new().on_blue());
226
227        StatefulWidget::render(table, area, buf, &mut state.table_state);
228    }
229}
