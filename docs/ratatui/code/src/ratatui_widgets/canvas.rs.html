1//! A [`Canvas`] and a collection of [`Shape`]s.
2//!

4//! predefined [`Shape`]s.

10//! - [`Map`]: A world map

66/// A single layer of the canvas.
67///
68/// This allows the canvas to be drawn in multiple layers. This is useful if you want to draw
69/// multiple shapes on the canvas in specific order.
70#[derive(Debug)]
71struct Layer {
72    // A string of characters representing the grid. This will be wrapped to the width of the grid
73    // when rendering
74    string: String,
75    // Colors for foreground and background of each cell
76    colors: Vec<(Color, Color)>,
77}
78
79/// A grid of cells that can be painted on.
80///
81/// The grid represents a particular screen region measured in rows and columns. The underlying
82/// resolution of the grid might exceed the number of rows and columns. For example, a grid of
83/// Braille patterns will have a resolution of 2x4 dots per cell. This means that a grid of 10x10
84/// cells will have a resolution of 20x40 dots.
85trait Grid: fmt::Debug {
86    /// Get the resolution of the grid in number of dots.
87    ///
88    /// This doesn't have to be the same as the number of rows and columns of the grid. For example,
89    /// a grid of Braille patterns will have a resolution of 2x4 dots per cell. This means that a
90    /// grid of 10x10 cells will have a resolution of 20x40 dots.
91    fn resolution(&self) -> (f64, f64);
92    /// Paint a point of the grid.
93    ///
94    /// The point is expressed in number of dots starting at the origin of the grid in the top left
95    /// corner. Note that this is not the same as the `(x, y)` coordinates of the canvas.
96    fn paint(&mut self, x: usize, y: usize, color: Color);
97    /// Save the current state of the [`Grid`] as a layer to be rendered
98    fn save(&self) -> Layer;
99    /// Reset the grid to its initial state
100    fn reset(&mut self);
101}

103/// The `BrailleGrid` is a grid made up of cells each containing a Braille pattern.
104///
105/// This makes it possible to draw shapes with a resolution of 2x4 dots per cell. This is useful
106/// when you want to draw shapes with a high resolution. Font support for Braille patterns is
107/// required to see the dots. If your terminal or font does not support this unicode block, you
108/// will see unicode replacement characters (�) instead of braille dots.
109///
110/// This grid type only supports a single foreground color for each 2x4 dots cell. There is no way
111/// to set the individual color of each dot in the braille pattern.
112#[derive(Debug)]
113struct BrailleGrid {
114    /// Width of the grid in number of terminal columns
115    width: u16,
116    /// Height of the grid in number of terminal rows
117    height: u16,
118    /// Represents the unicode braille patterns. Will take a value between `0x2800` and `0x28FF`
119    /// this is converted to an utf16 string when converting to a layer. See
120    /// <https://en.wikipedia.org/wiki/Braille_Patterns> for more info.
121    utf16_code_points: Vec<u16>,
122    /// The color of each cell only supports foreground colors for now as there's no way to
123    /// individually set the background color of each dot in the braille pattern.
124    colors: Vec<Color>,
125}
126
127impl BrailleGrid {
128    /// Create a new `BrailleGrid` with the given width and height measured in terminal columns and
129    /// rows respectively.
130    fn new(width: u16, height: u16) -> Self {
131        let length = usize::from(width) * usize::from(height);
132        Self {
133            width,
134            height,
135            utf16_code_points: vec![symbols::braille::BLANK; length],
136            colors: vec![Color::Reset; length],
137        }
138    }
139}
140
141impl Grid for BrailleGrid {
142    fn resolution(&self) -> (f64, f64) {
143        (f64::from(self.width) * 2.0, f64::from(self.height) * 4.0)
144    }
145
146    fn save(&self) -> Layer {
147        let string = String::from_utf16(&self.utf16_code_points).unwrap();
148        // the background color is always reset for braille patterns
149        let colors = self.colors.iter().map(|c| (*c, Color::Reset)).collect();
150        Layer { string, colors }
151    }
152
153    fn reset(&mut self) {
154        self.utf16_code_points.fill(symbols::braille::BLANK);
155        self.colors.fill(Color::Reset);
156    }
157
158    fn paint(&mut self, x: usize, y: usize, color: Color) {
159        let index = y
160            .saturating_div(4)
161            .saturating_mul(self.width as usize)
162            .saturating_add(x.saturating_div(2));
163        // using get_mut here because we are indexing the vector with usize values
164        // and we want to make sure we don't panic if the index is out of bounds
165        if let Some(c) = self.utf16_code_points.get_mut(index) {
166            *c |= symbols::braille::DOTS[y % 4][x % 2];
167        }
168        if let Some(c) = self.colors.get_mut(index) {
169            *c = color;
170        }
171    }
172}

174/// The `CharGrid` is a grid made up of cells each containing a single character.
175///
176/// This makes it possible to draw shapes with a resolution of 1x1 dots per cell. This is useful
177/// when you want to draw shapes with a low resolution.
178#[derive(Debug)]
179struct CharGrid {
180    /// Width of the grid in number of terminal columns
181    width: u16,
182    /// Height of the grid in number of terminal rows
183    height: u16,
184    /// Represents a single character for each cell
185    cells: Vec<char>,
186    /// The color of each cell
187    colors: Vec<Color>,
188    /// The character to use for every cell - e.g. a block, dot, etc.
189    cell_char: char,
190}
191
192impl CharGrid {
193    /// Create a new `CharGrid` with the given width and height measured in terminal columns and
194    /// rows respectively.
195    fn new(width: u16, height: u16, cell_char: char) -> Self {
196        let length = usize::from(width) * usize::from(height);
197        Self {
198            width,
199            height,
200            cells: vec![' '; length],
201            colors: vec![Color::Reset; length],
202            cell_char,
203        }
204    }
205}
206
207impl Grid for CharGrid {
208    fn resolution(&self) -> (f64, f64) {
209        (f64::from(self.width), f64::from(self.height))
210    }
211
212    fn save(&self) -> Layer {
213        Layer {
214            string: self.cells.iter().collect(),
215            colors: self.colors.iter().map(|c| (*c, Color::Reset)).collect(),
216        }
217    }
218
219    fn reset(&mut self) {
220        self.cells.fill(' ');
221        self.colors.fill(Color::Reset);
222    }
223
224    fn paint(&mut self, x: usize, y: usize, color: Color) {
225        let index = y.saturating_mul(self.width as usize).saturating_add(x);
226        // using get_mut here because we are indexing the vector with usize values
227        // and we want to make sure we don't panic if the index is out of bounds
228        if let Some(c) = self.cells.get_mut(index) {
229            *c = self.cell_char;
230        }
231        if let Some(c) = self.colors.get_mut(index) {
232            *c = color;
233        }
234    }
235}

237/// The `HalfBlockGrid` is a grid made up of cells each containing a half block character.
238///
239/// In terminals, each character is usually twice as tall as it is wide. Unicode has a couple of
240/// vertical half block characters, the upper half block '▀' and lower half block '▄' which take up
241/// half the height of a normal character but the full width. Together with an empty space ' ' and a
242/// full block '█', we can effectively double the resolution of a single cell. In addition, because
243/// each character can have a foreground and background color, we can control the color of the upper
244/// and lower half of each cell. This allows us to draw shapes with a resolution of 1x2 "pixels" per
245/// cell.
246///
247/// This allows for more flexibility than the `BrailleGrid` which only supports a single
248/// foreground color for each 2x4 dots cell, and the `CharGrid` which only supports a single
249/// character for each cell.
250#[derive(Debug)]
251struct HalfBlockGrid {
252    /// Width of the grid in number of terminal columns
253    width: u16,
254    /// Height of the grid in number of terminal rows
255    height: u16,
256    /// Represents a single color for each "pixel" arranged in column, row order
257    pixels: Vec<Vec<Color>>,
258}
259
260impl HalfBlockGrid {
261    /// Create a new `HalfBlockGrid` with the given width and height measured in terminal columns
262    /// and rows respectively.
263    fn new(width: u16, height: u16) -> Self {
264        Self {
265            width,
266            height,
267            pixels: vec![vec![Color::Reset; width as usize]; (height as usize) * 2],
268        }
269    }
270}
271
272impl Grid for HalfBlockGrid {
273    fn resolution(&self) -> (f64, f64) {
274        (f64::from(self.width), f64::from(self.height) * 2.0)
275    }
276
277    fn save(&self) -> Layer {
278        // Given that we store the pixels in a grid, and that we want to use 2 pixels arranged
279        // vertically to form a single terminal cell, which can be either empty, upper half block,
280        // lower half block or full block, we need examine the pixels in vertical pairs to decide
281        // what character to print in each cell. So these are the 4 states we use to represent each
282        // cell:
283        //
284        // 1. upper: reset, lower: reset => ' ' fg: reset / bg: reset
285        // 2. upper: reset, lower: color => '▄' fg: lower color / bg: reset
286        // 3. upper: color, lower: reset => '▀' fg: upper color / bg: reset
287        // 4. upper: color, lower: color => '▀' fg: upper color / bg: lower color
288        //
289        // Note that because the foreground reset color (i.e. default foreground color) is usually
290        // not the same as the background reset color (i.e. default background color), we need to
291        // swap around the colors for that state (2 reset/color).
292        //
293        // When the upper and lower colors are the same, we could continue to use an upper half
294        // block, but we choose to use a full block instead. This allows us to write unit tests that
295        // treat the cell as a single character instead of two half block characters.
296
297        // first we join each adjacent row together to get an iterator that contains vertical pairs
298        // of pixels, with the lower row being the first element in the pair
299        let vertical_color_pairs = self
300            .pixels
301            .iter()
302            .tuples()
303            .flat_map(|(upper_row, lower_row)| zip(upper_row, lower_row));
304
305        // then we work out what character to print for each pair of pixels
306        let string = vertical_color_pairs
307            .clone()
308            .map(|(upper, lower)| match (upper, lower) {
309                (Color::Reset, Color::Reset) => ' ',
310                (Color::Reset, _) => symbols::half_block::LOWER,
311                (_, Color::Reset) => symbols::half_block::UPPER,
312                (&lower, &upper) => {
313                    if lower == upper {
314                        symbols::half_block::FULL
315                    } else {
316                        symbols::half_block::UPPER
317                    }
318                }
319            })
320            .collect();
321
322        // then we convert these each vertical pair of pixels into a foreground and background color
323        let colors = vertical_color_pairs
324            .map(|(upper, lower)| {
325                let (fg, bg) = match (upper, lower) {
326                    (Color::Reset, Color::Reset) => (Color::Reset, Color::Reset),
327                    (Color::Reset, &lower) => (lower, Color::Reset),
328                    (&upper, Color::Reset) => (upper, Color::Reset),
329                    (&upper, &lower) => (upper, lower),
330                };
331                (fg, bg)
332            })
333            .collect();
334
335        Layer { string, colors }
336    }
337
338    fn reset(&mut self) {
339        self.pixels.fill(vec![Color::Reset; self.width as usize]);
340    }
341
342    fn paint(&mut self, x: usize, y: usize, color: Color) {
343        self.pixels[y][x] = color;
344    }
345}

347/// Painter is an abstraction over the [`Context`] that allows to draw shapes on the grid.
348///
349/// It is used by the [`Shape`] trait to draw shapes on the grid. It can be useful to think of this
350/// as similar to the [`Buffer`] struct that is used to draw widgets on the terminal.
351#[derive(Debug)]
352pub struct Painter<'a, 'b> {
353    context: &'a mut Context<'b>,
354    resolution: (f64, f64),
355}
356
357impl Painter<'_, '_> {
358    /// Convert the `(x, y)` coordinates to location of a point on the grid
359    ///
360    /// `(x, y)` coordinates are expressed in the coordinate system of the canvas. The origin is in
361    /// the lower left corner of the canvas (unlike most other coordinates in `Ratatui` where the
362    /// origin is the upper left corner). The `x` and `y` bounds of the canvas define the specific
363    /// area of some coordinate system that will be drawn on the canvas. The resolution of the grid
364    /// is used to convert the `(x, y)` coordinates to the location of a point on the grid.
365    ///
366    /// The grid coordinates are expressed in the coordinate system of the grid. The origin is in
367    /// the top left corner of the grid. The x and y bounds of the grid are always `[0, width - 1]`
368    /// and `[0, height - 1]` respectively. The resolution of the grid is used to convert the
369    /// `(x, y)` coordinates to the location of a point on the grid.
370    ///
371    /// Points are rounded to the nearest grid cell (with points exactly in the center of a cell
372    /// rounding up).
373    ///
374    /// # Examples
375    ///
376    /// ```
377    /// use ratatui::symbols;
378    /// use ratatui::widgets::canvas::{Context, Painter};
379    ///
380    /// let mut ctx = Context::new(2, 2, [1.0, 2.0], [0.0, 2.0], symbols::Marker::Braille);
381    /// let mut painter = Painter::from(&mut ctx);
382    ///
383    /// let point = painter.get_point(1.0, 0.0);
384    /// assert_eq!(point, Some((0, 7)));
385    ///
386    /// let point = painter.get_point(1.5, 1.0);
387    /// assert_eq!(point, Some((2, 4)));
388    ///
389    /// let point = painter.get_point(0.0, 0.0);
390    /// assert_eq!(point, None);
391    ///
392    /// let point = painter.get_point(2.0, 2.0);
393    /// assert_eq!(point, Some((3, 0)));
394    ///
395    /// let point = painter.get_point(1.0, 2.0);
396    /// assert_eq!(point, Some((0, 0)));
397    /// ```
398    pub fn get_point(&self, x: f64, y: f64) -> Option<(usize, usize)> {
399        let [left, right] = self.context.x_bounds;
400        let [bottom, top] = self.context.y_bounds;
401        if x < left || x > right || y < bottom || y > top {
402            return None;
403        }
404        let width = right - left;
405        let height = top - bottom;
406        if width <= 0.0 || height <= 0.0 {
407            return None;
408        }
409        let x = ((x - left) * (self.resolution.0 - 1.0) / width).round() as usize;
410        let y = ((top - y) * (self.resolution.1 - 1.0) / height).round() as usize;
411        Some((x, y))
412    }
413
414    /// Paint a point of the grid
415    ///
416    /// # Example
417    ///
418    /// ```
419    /// use ratatui::style::Color;
420    /// use ratatui::symbols;
421    /// use ratatui::widgets::canvas::{Context, Painter};
422    ///
423    /// let mut ctx = Context::new(1, 1, [0.0, 2.0], [0.0, 2.0], symbols::Marker::Braille);
424    /// let mut painter = Painter::from(&mut ctx);
425    /// painter.paint(1, 3, Color::Red);
426    /// ```
427    pub fn paint(&mut self, x: usize, y: usize, color: Color) {
428        self.context.grid.paint(x, y, color);
429    }
430
431    /// Canvas context bounds by axis.
432    ///
433    /// # Example
434    ///
435    /// ```
436    /// use ratatui::style::Color;
437    /// use ratatui::symbols;
438    /// use ratatui::widgets::canvas::{Context, Painter};
439    ///
440    /// let mut ctx = Context::new(1, 1, [0.0, 2.0], [0.0, 2.0], symbols::Marker::Braille);
441    /// let mut painter = Painter::from(&mut ctx);
442    /// assert_eq!(painter.bounds(), (&[0.0, 2.0], &[0.0, 2.0]));
443    /// ```
444    pub const fn bounds(&self) -> (&[f64; 2], &[f64; 2]) {
445        (&self.context.x_bounds, &self.context.y_bounds)
446    }
447}

452        Self {

454            resolution,
455        }

459/// Holds the state of the [`Canvas`] when painting to it.
460///
461/// This is used by the [`Canvas`] widget to draw shapes on the grid. It can be useful to think of
462/// this as similar to the `Frame` struct that is used to draw widgets on the terminal.
463#[derive(Debug)]
464pub struct Context<'a> {
465    x_bounds: [f64; 2],
466    y_bounds: [f64; 2],
467    grid: Box<dyn Grid>,
468    dirty: bool,
469    layers: Vec<Layer>,
470    labels: Vec<Label<'a>>,
471}
472
473impl<'a> Context<'a> {
474    /// Create a new Context with the given width and height measured in terminal columns and rows
475    /// respectively. The `x` and `y` bounds define the specific area of some coordinate system that
476    /// will be drawn on the canvas. The marker defines the type of points used to draw the shapes.
477    ///
478    /// Applications should not use this directly but rather use the [`Canvas`] widget. This will be
479    /// created by the [`Canvas::paint`] method and passed to the closure that is used to draw on
480    /// the canvas.
481    ///
482    /// The `x` and `y` bounds should be specified as left/right and bottom/top respectively. For
483    /// example, if you want to draw a map of the world, you might want to use the following bounds:
484    ///
485    /// ```
486    /// use ratatui::symbols;
487    /// use ratatui::widgets::canvas::Context;
488    ///
489    /// let ctx = Context::new(
490    ///     100,
491    ///     100,
492    ///     [-180.0, 180.0],
493    ///     [-90.0, 90.0],
494    ///     symbols::Marker::Braille,
495    /// );
496    /// ```
497    pub fn new(
498        width: u16,
499        height: u16,
500        x_bounds: [f64; 2],
501        y_bounds: [f64; 2],
502        marker: Marker,
503    ) -> Self {
504        let dot = symbols::DOT.chars().next().unwrap();
505        let block = symbols::block::FULL.chars().next().unwrap();
506        let bar = symbols::bar::HALF.chars().next().unwrap();
507        let grid: Box<dyn Grid> = match marker {
508            Marker::Dot => Box::new(CharGrid::new(width, height, dot)),
509            Marker::Block => Box::new(CharGrid::new(width, height, block)),
510            Marker::Bar => Box::new(CharGrid::new(width, height, bar)),
511            Marker::Braille => Box::new(BrailleGrid::new(width, height)),
512            Marker::HalfBlock => Box::new(HalfBlockGrid::new(width, height)),
513        };
514        Self {
515            x_bounds,
516            y_bounds,
517            grid,
518            dirty: false,
519            layers: Vec::new(),
520            labels: Vec::new(),
521        }
522    }
523
524    /// Draw the given [`Shape`] in this context
525    pub fn draw<S>(&mut self, shape: &S)
526    where
527        S: Shape,
528    {
529        self.dirty = true;
530        let mut painter = Painter::from(self);
531        shape.draw(&mut painter);
532    }
533
534    /// Save the existing state of the grid as a layer.
535    ///
536    /// Save the existing state as a layer to be rendered and reset the grid to its initial
537    /// state for the next layer.
538    ///
539    /// This allows the canvas to be drawn in multiple layers. This is useful if you want to
540    /// draw multiple shapes on the [`Canvas`] in specific order.
541    pub fn layer(&mut self) {
542        self.layers.push(self.grid.save());
543        self.grid.reset();
544        self.dirty = false;
545    }
546
547    /// Print a [`Text`] on the [`Canvas`] at the given position.
548    ///
549    /// Note that the text is always printed on top of the canvas and is **not** affected by the
550    /// layers.
551    ///
552    /// [`Text`]: ratatui_core::text::Text
553    pub fn print<T>(&mut self, x: f64, y: f64, line: T)
554    where
555        T: Into<TextLine<'a>>,
556    {
557        self.labels.push(Label {
558            x,
559            y,
560            line: line.into(),
561        });
562    }
563
564    /// Save the last layer if necessary
565    fn finish(&mut self) {
566        if self.dirty {
567            self.layer();
568        }
569    }
570}

572/// The Canvas widget provides a means to draw shapes (Lines, Rectangles, Circles, etc.) on a grid.
573///
574/// By default the grid is made of Braille patterns but you may change the marker to use a different
575/// set of symbols. If your terminal or font does not support this unicode block, you will see
576/// unicode replacement characters (�) instead of braille dots. The Braille patterns provide a more
577/// fine grained result (2x4 dots) but you might want to use a simple dot, block, or bar instead by
578/// calling the [`marker`] method if your target environment does not support those symbols.
579///
580/// See [Unicode Braille Patterns](https://en.wikipedia.org/wiki/Braille_Patterns) for more info.
581///
582/// The `HalfBlock` marker is useful when you want to draw shapes with a higher resolution than a
583/// `CharGrid` but lower than a `BrailleGrid`. This grid type supports a foreground and background
584/// color for each terminal cell. This allows for more flexibility than the `BrailleGrid` which only
585/// supports a single foreground color for each 2x4 dots cell.
586///
587/// The Canvas widget is used by calling the [`Canvas::paint`] method and passing a closure that
588/// will be used to draw on the canvas. The closure will be passed a [`Context`] object that can be
589/// used to draw shapes on the canvas.
590///
591/// The [`Context`] object provides a [`Context::draw`] method that can be used to draw shapes on
592/// the canvas. The [`Context::layer`] method can be used to save the current state of the canvas
593/// and start a new layer. This is useful if you want to draw multiple shapes on the canvas in
594/// specific order. The [`Context`] object also provides a [`Context::print`] method that can be
595/// used to print text on the canvas. Note that the text is always printed on top of the canvas and
596/// is not affected by the layers.
597///
598/// # Examples
599///
600/// ```
601/// use ratatui::style::Color;
602/// use ratatui::widgets::Block;
603/// use ratatui::widgets::canvas::{Canvas, Line, Map, MapResolution, Rectangle};
604///
605/// Canvas::default()
606///     .block(Block::bordered().title("Canvas"))
607///     .x_bounds([-180.0, 180.0])
608///     .y_bounds([-90.0, 90.0])
609///     .paint(|ctx| {
610///         ctx.draw(&Map {
611///             resolution: MapResolution::High,
612///             color: Color::White,
613///         });
614///         ctx.layer();
615///         ctx.draw(&Line {
616///             x1: 0.0,
617///             y1: 10.0,
618///             x2: 10.0,
619///             y2: 10.0,
620///             color: Color::White,
621///         });
622///         ctx.draw(&Rectangle {
623///             x: 10.0,
624///             y: 20.0,
625///             width: 10.0,
626///             height: 10.0,
627///             color: Color::Red,
628///         });
629///     });
630/// ```
631///
632/// [`marker`]: #method.marker
633#[derive(Debug, Clone, PartialEq)]
634pub struct Canvas<'a, F>
635where
636    F: Fn(&mut Context),
637{
638    block: Option<Block<'a>>,
639    x_bounds: [f64; 2],
640    y_bounds: [f64; 2],
641    paint_func: Option<F>,
642    background_color: Color,
643    marker: Marker,
644}

708    /// Change the background [`Color`] of the entire canvas

713        self.background_color = color;
714        self
715    }
716
717    /// Change the type of points used to draw the shapes.
718    ///
719    /// By default the [`Braille`] patterns are used as they provide a more fine grained result,
720    /// but you might want to use the simple [`Dot`] or [`Block`] instead if the targeted terminal
721    /// does not support those symbols.
722    ///
723    /// The [`HalfBlock`] marker is useful when you want to draw shapes with a higher resolution
724    /// than with a grid of characters (e.g. with [`Block`] or [`Dot`]) but lower than with
725    /// [`Braille`]. This grid type supports a foreground and background color for each terminal
726    /// cell. This allows for more flexibility than the `BrailleGrid` which only supports a single
727    /// foreground color for each 2x4 dots cell.
728    ///
729    /// [`Braille`]: ratatui_core::symbols::Marker::Braille
730    /// [`HalfBlock`]: ratatui_core::symbols::Marker::HalfBlock
731    /// [`Dot`]: ratatui_core::symbols::Marker::Dot
732    /// [`Block`]: ratatui_core::symbols::Marker::Block
733    ///
734    /// # Examples
735    ///
736    /// ```
737    /// use ratatui::symbols;
738    /// use ratatui::widgets::canvas::Canvas;
739    ///
740    /// Canvas::default()
741    ///     .marker(symbols::Marker::Braille)
742    ///     .paint(|ctx| {});
743    ///
744    /// Canvas::default()
745    ///     .marker(symbols::Marker::HalfBlock)
746    ///     .paint(|ctx| {});
747    ///
748    /// Canvas::default()
749    ///     .marker(symbols::Marker::Dot)
750    ///     .paint(|ctx| {});
751    ///
752    /// Canvas::default()
753    ///     .marker(symbols::Marker::Block)
754    ///     .paint(|ctx| {});
755    /// ```
756    #[must_use = "method moves the value of self and returns the modified value"]
757    pub const fn marker(mut self, marker: Marker) -> Self {
758        self.marker = marker;
759        self
760    }
761}

763impl<F> Widget for Canvas<'_, F>
764where
765    F: Fn(&mut Context),
766{
767    fn render(self, area: Rect, buf: &mut Buffer) {
768        Widget::render(&self, area, buf);
769    }
770}
771
772impl<F> Widget for &Canvas<'_, F>
773where
774    F: Fn(&mut Context),
775{
776    fn render(self, area: Rect, buf: &mut Buffer) {
777        self.block.as_ref().render(area, buf);
778        let canvas_area = self.block.inner_if_some(area);
779        if canvas_area.is_empty() {
780            return;
781        }
782
783        buf.set_style(canvas_area, Style::default().bg(self.background_color));
784
785        let width = canvas_area.width as usize;
786
787        let Some(ref painter) = self.paint_func else {
788            return;
789        };
790
791        // Create a blank context that match the size of the canvas
792        let mut ctx = Context::new(
793            canvas_area.width,
794            canvas_area.height,
795            self.x_bounds,
796            self.y_bounds,
797            self.marker,
798        );
799        // Paint to this context
800        painter(&mut ctx);
801        ctx.finish();
802
803        // Retrieve painted points for each layer
804        for layer in ctx.layers {
805            for (index, (ch, colors)) in layer.string.chars().zip(layer.colors).enumerate() {
806                if ch != ' ' && ch != '\u{2800}' {
807                    let (x, y) = (
808                        (index % width) as u16 + canvas_area.left(),
809                        (index / width) as u16 + canvas_area.top(),
810                    );
811                    let cell = buf[(x, y)].set_char(ch);
812                    if colors.0 != Color::Reset {
813                        cell.set_fg(colors.0);
814                    }
815                    if colors.1 != Color::Reset {
816                        cell.set_bg(colors.1);
817                    }
818                }
819            }
820        }
821
822        // Finally draw the labels
823        let left = self.x_bounds[0];
824        let right = self.x_bounds[1];
825        let top = self.y_bounds[1];
826        let bottom = self.y_bounds[0];
827        let width = (self.x_bounds[1] - self.x_bounds[0]).abs();
828        let height = (self.y_bounds[1] - self.y_bounds[0]).abs();
829        let resolution = {
830            let width = f64::from(canvas_area.width - 1);
831            let height = f64::from(canvas_area.height - 1);
832            (width, height)
833        };
834        for label in ctx
835            .labels
836            .iter()
837            .filter(|l| l.x >= left && l.x <= right && l.y <= top && l.y >= bottom)
838        {
839            let x = ((label.x - left) * resolution.0 / width) as u16 + canvas_area.left();
840            let y = ((top - label.y) * resolution.1 / height) as u16 + canvas_area.top();
841            buf.set_line(x, y, &label.line, canvas_area.right() - x);
842        }
843    }
844}
