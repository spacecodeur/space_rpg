1//! A Ratatui example that shows the full range of RGB colors that can be displayed in the terminal.
2//!
3//! Requires a terminal that supports 24-bit color (true color) and unicode.
4//!
5//! This example also demonstrates how implementing the Widget trait on a mutable reference
6//! allows the widget to update its state while it is being rendered. This allows the fps
7//! widget to update the fps calculation and the colors widget to update a cached version of
8//! the colors to render instead of recalculating them every frame.
9//!
10//! This is an alternative to using the `StatefulWidget` trait and a separate state struct. It
11//! is useful when the state is only used by the widget and doesn't need to be shared with
12//! other widgets.
13//!
14//! This example runs with the Ratatui library code in the branch that you are currently reading.
15//! See the [`latest`] branch for the code which works with the most recent Ratatui release.
16//!
17//! [`latest`]: https://github.com/ratatui/ratatui/tree/latest
18

32fn main() -> Result<()> {
33    color_eyre::install()?;
34    ratatui::run(|terminal| App::default().run(terminal))
35}

37#[derive(Debug, Default)]
38struct App {
39    /// The current state of the app (running or quit)
40    state: AppState,
41
42    /// A widget that displays the current frames per second
43    fps_widget: FpsWidget,
44
45    /// A widget that displays the full range of RGB colors that can be displayed in the terminal.
46    colors_widget: ColorsWidget,
47}
48
49#[derive(Debug, Default, PartialEq, Eq)]
50enum AppState {
51    /// The app is running
52    #[default]
53    Running,
54
55    /// The user has requested the app to quit
56    Quit,
57}
58
59/// A widget that displays the current frames per second
60#[derive(Debug)]
61struct FpsWidget {
62    /// The number of elapsed frames that have passed - used to calculate the fps
63    frame_count: usize,
64
65    /// The last instant that the fps was calculated
66    last_instant: Instant,
67
68    /// The current frames per second
69    fps: Option<f32>,
70}
71
72/// A widget that displays the full range of RGB colors that can be displayed in the terminal.
73///
74/// This widget is animated and will change colors over time.
75#[derive(Debug, Default)]
76struct ColorsWidget {
77    /// The colors to render - should be double the height of the area as we render two rows of
78    /// pixels for each row of the widget using the half block character. This is computed any time
79    /// the size of the widget changes.
80    colors: Vec<Vec<Color>>,
81
82    /// the number of elapsed frames that have passed - used to animate the colors by shifting the
83    /// x index by the frame number
84    frame_count: usize,
85}
86
87impl App {
88    /// Run the app
89    ///
90    /// This is the main event loop for the app.
91    pub fn run(mut self, terminal: &mut DefaultTerminal) -> Result<()> {
92        while self.is_running() {
93            terminal.draw(|frame| frame.render_widget(&mut self, frame.area()))?;
94            self.handle_events()?;
95        }
96        Ok(())
97    }
98
99    const fn is_running(&self) -> bool {
100        matches!(self.state, AppState::Running)
101    }
102
103    /// Handle any events that have occurred since the last time the app was rendered.
104    fn handle_events(&mut self) -> Result<()> {
105        // Ensure that the app only blocks for a period that allows the app to render at
106        // approximately 60 FPS (this doesn't account for the time to render the frame, and will
107        // also update the app immediately any time an event occurs)
108        let timeout = Duration::from_secs_f32(1.0 / 60.0);
109        if !event::poll(timeout)? {
110            return Ok(());
111        }
112        if event::read()?.is_key_press() {
113            self.state = AppState::Quit;
114        }
115        Ok(())
116    }
117}

119/// Implement the Widget trait for &mut App so that it can be rendered
120///
121/// This is implemented on a mutable reference so that the app can update its state while it is
122/// being rendered. This allows the fps widget to update the fps calculation and the colors widget
123/// to update the colors to render.
124impl Widget for &mut App {
125    fn render(self, area: Rect, buf: &mut Buffer) {
126        use Constraint::{Length, Min};
127        let [top, colors] = area.layout(&Layout::vertical([Length(1), Min(0)]));
128        let [title, fps] = top.layout(&Layout::horizontal([Min(0), Length(8)]));
129        Text::from("colors_rgb example. Press q to quit")
130            .centered()
131            .render(title, buf);
132        self.fps_widget.render(fps, buf);
133        self.colors_widget.render(colors, buf);
134    }
135}

137/// Default impl for `FpsWidget`
138///
139/// Manual impl is required because we need to initialize the `last_instant` field to the current
140/// instant.
141impl Default for FpsWidget {
142    fn default() -> Self {
143        Self {
144            frame_count: 0,
145            last_instant: Instant::now(),
146            fps: None,
147        }
148    }
149}
150
151/// Widget impl for `FpsWidget`
152///
153/// This is implemented on a mutable reference so that we can update the frame count and fps
154/// calculation while rendering.
155impl Widget for &mut FpsWidget {
156    fn render(self, area: Rect, buf: &mut Buffer) {
157        self.calculate_fps();
158        if let Some(fps) = self.fps {
159            let text = format!("{fps:.1} fps");
160            Text::from(text).render(area, buf);
161        }
162    }
163}
164
165impl FpsWidget {
166    /// Update the fps calculation.
167    ///
168    /// This updates the fps once a second, but only if the widget has rendered at least 2 frames
169    /// since the last calculation. This avoids noise in the fps calculation when rendering on slow
170    /// machines that can't render at least 2 frames per second.
171    #[expect(clippy::cast_precision_loss)]
172    fn calculate_fps(&mut self) {
173        self.frame_count += 1;
174        let elapsed = self.last_instant.elapsed();
175        if elapsed > Duration::from_secs(1) && self.frame_count > 2 {
176            self.fps = Some(self.frame_count as f32 / elapsed.as_secs_f32());
177            self.frame_count = 0;
178            self.last_instant = Instant::now();
179        }
180    }
181}

183/// Widget impl for `ColorsWidget`
184///
185/// This is implemented on a mutable reference so that we can update the frame count and store a
186/// cached version of the colors to render instead of recalculating them every frame.
187impl Widget for &mut ColorsWidget {
188    /// Render the widget
189    fn render(self, area: Rect, buf: &mut Buffer) {
190        self.setup_colors(area);
191        let colors = &self.colors;
192        for (xi, x) in (area.left()..area.right()).enumerate() {
193            // animate the colors by shifting the x index by the frame number
194            let xi = (xi + self.frame_count) % (area.width as usize);
195            for (yi, y) in (area.top()..area.bottom()).enumerate() {
196                // render a half block character for each row of pixels with the foreground color
197                // set to the color of the pixel and the background color set to the color of the
198                // pixel below it
199                let fg = colors[yi * 2][xi];
200                let bg = colors[yi * 2 + 1][xi];
201                buf[Position::new(x, y)].set_char('▀').set_fg(fg).set_bg(bg);
202            }
203        }
204        self.frame_count += 1;
205    }
206}
207
208impl ColorsWidget {
209    /// Setup the colors to render.
210    ///
211    /// This is called once per frame to setup the colors to render. It caches the colors so that
212    /// they don't need to be recalculated every frame.
213    #[expect(clippy::cast_precision_loss)]
214    fn setup_colors(&mut self, size: Rect) {
215        let Rect { width, height, .. } = size;
216        // double the height because each screen row has two rows of half block pixels
217        let height = height as usize * 2;
218        let width = width as usize;
219        // only update the colors if the size has changed since the last time we rendered
220        if self.colors.len() == height && self.colors[0].len() == width {
221            return;
222        }
223        self.colors = Vec::with_capacity(height);
224        for y in 0..height {
225            let mut row = Vec::with_capacity(width);
226            for x in 0..width {
227                let hue = x as f32 * 360.0 / width as f32;
228                let value = (height - y) as f32 / height as f32;
229                let saturation = Okhsv::max_saturation();
230                let color = Okhsv::new(hue, saturation, value);
231                let color = Srgb::<f32>::from_color_unclamped(color);
232                let color: Srgb<u8> = color.into_format();
233                let color = Color::Rgb(color.red, color.green, color.blue);
234                row.push(color);
235            }
236            self.colors.push(row);
237        }
238    }
