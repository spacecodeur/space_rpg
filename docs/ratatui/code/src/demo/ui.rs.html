13pub fn render(frame: &mut Frame, app: &mut App) {
14    let chunks = Layout::vertical([Constraint::Length(3), Constraint::Min(0)]).split(frame.area());
15    let tabs = app
16        .tabs
17        .titles
18        .iter()
19        .map(|t| text::Line::from(Span::styled(*t, Style::default().fg(Color::Green))))
20        .collect::<Tabs>()
21        .block(Block::bordered().title(app.title))
22        .highlight_style(Style::default().fg(Color::Yellow))
23        .select(app.tabs.index);
24    frame.render_widget(tabs, chunks[0]);
25    match app.tabs.index {
26        0 => draw_first_tab(frame, app, chunks[1]),
27        1 => draw_second_tab(frame, app, chunks[1]),
28        2 => draw_third_tab(frame, app, chunks[1]),
29        _ => {}
30    };
31}

45fn draw_gauges(frame: &mut Frame, app: &mut App, area: Rect) {
46    let chunks = Layout::vertical([
47        Constraint::Length(2),
48        Constraint::Length(3),
49        Constraint::Length(2),
50    ])
51    .margin(1)
52    .split(area);
53    let block = Block::bordered().title("Graphs");
54    frame.render_widget(block, area);
55
56    let label = format!("{:.2}%", app.progress * 100.0);
57    let gauge = Gauge::default()
58        .block(Block::new().title("Gauge:"))
59        .gauge_style(
60            Style::default()
61                .fg(Color::Magenta)
62                .bg(Color::Black)
63                .add_modifier(Modifier::ITALIC | Modifier::BOLD),
64        )
65        .use_unicode(app.enhanced_graphics)
66        .label(label)
67        .ratio(app.progress);
68    frame.render_widget(gauge, chunks[0]);
69
70    let sparkline = Sparkline::default()
71        .block(Block::new().title("Sparkline:"))
72        .style(Style::default().fg(Color::Green))
73        .data(&app.sparkline.points)
74        .bar_set(if app.enhanced_graphics {
75            symbols::bar::NINE_LEVELS
76        } else {
77            symbols::bar::THREE_LEVELS
78        });
79    frame.render_widget(sparkline, chunks[1]);
80
81    let line_gauge = LineGauge::default()
82        .block(Block::new().title("LineGauge:"))
83        .filled_style(Style::default().fg(Color::Magenta))
84        .filled_symbol(if app.enhanced_graphics {
85            symbols::line::THICK_HORIZONTAL
86        } else {
87            symbols::line::HORIZONTAL
88        })
89        .unfilled_symbol(if app.enhanced_graphics {
90            symbols::line::THICK_HORIZONTAL
91        } else {
92            symbols::line::HORIZONTAL
93        })
94        .ratio(app.progress);
95    frame.render_widget(line_gauge, chunks[2]);
96}

99fn draw_charts(frame: &mut Frame, app: &mut App, area: Rect) {
100    let constraints = if app.show_chart {
101        vec![Constraint::Percentage(50), Constraint::Percentage(50)]
102    } else {
103        vec![Constraint::Percentage(100)]
104    };
105    let chunks = Layout::horizontal(constraints).split(area);
106    {
107        let chunks = Layout::vertical([Constraint::Percentage(50), Constraint::Percentage(50)])
108            .split(chunks[0]);
109        {
110            let chunks =
111                Layout::horizontal([Constraint::Percentage(50), Constraint::Percentage(50)])
112                    .split(chunks[0]);
113
114            // Draw tasks
115            let tasks: Vec<ListItem> = app
116                .tasks
117                .items
118                .iter()
119                .map(|i| ListItem::new(vec![text::Line::from(Span::raw(*i))]))
120                .collect();
121            let tasks = List::new(tasks)
122                .block(Block::bordered().title("List"))
123                .highlight_style(Style::default().add_modifier(Modifier::BOLD))
124                .highlight_symbol("> ");
125            frame.render_stateful_widget(tasks, chunks[0], &mut app.tasks.state);
126
127            // Draw logs
128            let info_style = Style::default().fg(Color::Blue);
129            let warning_style = Style::default().fg(Color::Yellow);
130            let error_style = Style::default().fg(Color::Magenta);
131            let critical_style = Style::default().fg(Color::Red);
132            let logs: Vec<ListItem> = app
133                .logs
134                .items
135                .iter()
136                .map(|&(evt, level)| {
137                    let s = match level {
138                        "ERROR" => error_style,
139                        "CRITICAL" => critical_style,
140                        "WARNING" => warning_style,
141                        _ => info_style,
142                    };
143                    let content = vec![text::Line::from(vec![
144                        Span::styled(format!("{level:<9}"), s),
145                        Span::raw(evt),
146                    ])];
147                    ListItem::new(content)
148                })
149                .collect();
150            let logs = List::new(logs).block(Block::bordered().title("List"));
151            frame.render_stateful_widget(logs, chunks[1], &mut app.logs.state);
152        }
153
154        let barchart = BarChart::default()
155            .block(Block::bordered().title("Bar chart"))
156            .data(&app.barchart)
157            .bar_width(3)
158            .bar_gap(2)
159            .bar_set(if app.enhanced_graphics {
160                symbols::bar::NINE_LEVELS
161            } else {
162                symbols::bar::THREE_LEVELS
163            })
164            .value_style(
165                Style::default()
166                    .fg(Color::Black)
167                    .bg(Color::Green)
168                    .add_modifier(Modifier::ITALIC),
169            )
170            .label_style(Style::default().fg(Color::Yellow))
171            .bar_style(Style::default().fg(Color::Green));
172        frame.render_widget(barchart, chunks[1]);
173    }
174    if app.show_chart {
175        let x_labels = vec![
176            Span::styled(
177                format!("{}", app.signals.window[0]),
178                Style::default().add_modifier(Modifier::BOLD),
179            ),
180            Span::raw(format!(
181                "{}",
182                (app.signals.window[0] + app.signals.window[1]) / 2.0
183            )),
184            Span::styled(
185                format!("{}", app.signals.window[1]),
186                Style::default().add_modifier(Modifier::BOLD),
187            ),
188        ];
189        let datasets = vec![
190            Dataset::default()
191                .name("data2")
192                .marker(symbols::Marker::Dot)
193                .style(Style::default().fg(Color::Cyan))
194                .data(&app.signals.sin1.points),
195            Dataset::default()
196                .name("data3")
197                .marker(if app.enhanced_graphics {
198                    symbols::Marker::Braille
199                } else {
200                    symbols::Marker::Dot
201                })
202                .style(Style::default().fg(Color::Yellow))
203                .data(&app.signals.sin2.points),
204        ];
205        let chart = Chart::new(datasets)
206            .block(
207                Block::bordered().title(Span::styled(
208                    "Chart",
209                    Style::default()
210                        .fg(Color::Cyan)
211                        .add_modifier(Modifier::BOLD),
212                )),
213            )
214            .x_axis(
215                Axis::default()
216                    .title("X Axis")
217                    .style(Style::default().fg(Color::Gray))
218                    .bounds(app.signals.window)
219                    .labels(x_labels),
220            )
221            .y_axis(
222                Axis::default()
223                    .title("Y Axis")
224                    .style(Style::default().fg(Color::Gray))
225                    .bounds([-20.0, 20.0])
226                    .labels([
227                        Span::styled("-20", Style::default().add_modifier(Modifier::BOLD)),
228                        Span::raw("0"),
229                        Span::styled("20", Style::default().add_modifier(Modifier::BOLD)),
230                    ]),
231            );
232        frame.render_widget(chart, chunks[1]);
233    }
234}

277fn draw_second_tab(frame: &mut Frame, app: &mut App, area: Rect) {
278    let chunks =
279        Layout::horizontal([Constraint::Percentage(30), Constraint::Percentage(70)]).split(area);
280    let up_style = Style::default().fg(Color::Green);
281    let failure_style = Style::default()
282        .fg(Color::Red)
283        .add_modifier(Modifier::RAPID_BLINK | Modifier::CROSSED_OUT);
284    let rows = app.servers.iter().map(|s| {
285        let style = if s.status == "Up" {
286            up_style
287        } else {
288            failure_style
289        };
290        Row::new(vec![s.name, s.location, s.status]).style(style)
291    });
292    let table = Table::new(
293        rows,
294        [
295            Constraint::Length(15),
296            Constraint::Length(15),
297            Constraint::Length(10),
298        ],
299    )
300    .header(
301        Row::new(vec!["Server", "Location", "Status"])
302            .style(Style::default().fg(Color::Yellow))
303            .bottom_margin(1),
304    )
305    .block(Block::bordered().title("Servers"));
306    frame.render_widget(table, chunks[0]);
307
308    let map = Canvas::default()
309        .block(Block::bordered().title("World"))
310        .paint(|ctx| {
311            ctx.draw(&Map {
312                color: Color::White,
313                resolution: MapResolution::High,
314            });
315            ctx.layer();
316            ctx.draw(&Rectangle {
317                x: 0.0,
318                y: 30.0,
319                width: 10.0,
320                height: 10.0,
321                color: Color::Yellow,
322            });
323            ctx.draw(&Circle {
324                x: app.servers[2].coords.1,
325                y: app.servers[2].coords.0,
326                radius: 10.0,
327                color: Color::Green,
328            });
329            for (i, s1) in app.servers.iter().enumerate() {
330                for s2 in &app.servers[i + 1..] {
331                    ctx.draw(&canvas::Line {
332                        x1: s1.coords.1,
333                        y1: s1.coords.0,
334                        y2: s2.coords.0,
335                        x2: s2.coords.1,
336                        color: Color::Yellow,
337                    });
338                }
339            }
340            for server in &app.servers {
341                let color = if server.status == "Up" {
342                    Color::Green
343                } else {
344                    Color::Red
345                };
346                ctx.print(
347                    server.coords.1,
348                    server.coords.0,
349                    Span::styled("X", Style::default().fg(color)),
350                );
351            }
352        })
353        .marker(if app.enhanced_graphics {
354            symbols::Marker::Braille
355        } else {
356            symbols::Marker::Dot
357        })
358        .x_bounds([-180.0, 180.0])
359        .y_bounds([-90.0, 90.0]);
360    frame.render_widget(map, chunks[1]);
361}

363fn draw_third_tab(frame: &mut Frame, _app: &mut App, area: Rect) {
364    let chunks = Layout::horizontal([Constraint::Ratio(1, 2), Constraint::Ratio(1, 2)]).split(area);
365    let colors = [
366        Color::Reset,
367        Color::Black,
368        Color::Red,
369        Color::Green,
370        Color::Yellow,
371        Color::Blue,
372        Color::Magenta,
373        Color::Cyan,
374        Color::Gray,
375        Color::DarkGray,
376        Color::LightRed,
377        Color::LightGreen,
378        Color::LightYellow,
379        Color::LightBlue,
380        Color::LightMagenta,
381        Color::LightCyan,
382        Color::White,
383    ];
384    let items: Vec<Row> = colors
385        .iter()
386        .map(|c| {
387            let cells = vec![
388                Cell::from(Span::raw(format!("{c:?}: "))),
389                Cell::from(Span::styled("Foreground", Style::default().fg(*c))),
390                Cell::from(Span::styled("Background", Style::default().bg(*c))),
391            ];
392            Row::new(cells)
393        })
394        .collect();
395    let table = Table::new(
396        items,
397        [
398            Constraint::Ratio(1, 3),
399            Constraint::Ratio(1, 3),
400            Constraint::Ratio(1, 3),
401        ],
402    )
403    .block(Block::bordered().title("Colors"));
404    frame.render_widget(table, chunks[0]);
