1/// A simple example demonstrating how to handle user input.
2///
3/// This is a bit out of the scope of
4/// the library as it does not provide any input handling out of the box. However, it may helps
5/// some to get started.
6///
7/// This is a very simple example:
8///   * An input box always focused. Every character you type is registered here.
9///   * An entered character is inserted at the cursor position.
10///   * Pressing Backspace erases the left character before the cursor position
11///   * Pressing Enter pushes the current input in the history of previous messages.
12///
13/// **Note:** as this is a relatively simple example unicode characters are unsupported and
14/// their use will result in undefined behaviour.
15///
16/// See also <https://github.com/rhysd/tui-textarea> and <https://github.com/sayanarijit/tui-input>/
17///
18/// This example runs with the Ratatui library code in the branch that you are currently
19/// reading. See the [`latest`] branch for the code which works with the most recent Ratatui
20/// release.
21///
22/// [`latest`]: https://github.com/ratatui/ratatui/tree/latest
23use color_eyre::Result;
24use crossterm::event::{self, KeyCode, KeyEventKind};
25use ratatui::layout::{Constraint, Layout, Position};
26use ratatui::style::{Color, Modifier, Style, Stylize};
27use ratatui::text::{Line, Span, Text};
28use ratatui::widgets::{Block, List, ListItem, Paragraph};
29use ratatui::{DefaultTerminal, Frame};
30
31fn main() -> Result<()> {
32    color_eyre::install()?;
33    ratatui::run(|terminal| App::new().run(terminal))
34}
35
36/// App holds the state of the application
37struct App {
38    /// Current value of the input box
39    input: String,
40    /// Position of cursor in the editor area.
41    character_index: usize,
42    /// Current input mode
43    input_mode: InputMode,
44    /// History of recorded messages
45    messages: Vec<String>,
46}
47
48enum InputMode {
49    Normal,
50    Editing,
51}
52
53impl App {
54    const fn new() -> Self {
55        Self {
56            input: String::new(),
57            input_mode: InputMode::Normal,
58            messages: Vec::new(),
59            character_index: 0,
60        }
61    }
62
63    fn move_cursor_left(&mut self) {
64        let cursor_moved_left = self.character_index.saturating_sub(1);
65        self.character_index = self.clamp_cursor(cursor_moved_left);
66    }
67
68    fn move_cursor_right(&mut self) {
69        let cursor_moved_right = self.character_index.saturating_add(1);
70        self.character_index = self.clamp_cursor(cursor_moved_right);
71    }
72
73    fn enter_char(&mut self, new_char: char) {
74        let index = self.byte_index();
75        self.input.insert(index, new_char);
76        self.move_cursor_right();
77    }
78
79    /// Returns the byte index based on the character position.
80    ///
81    /// Since each character in a string can contain multiple bytes, it's necessary to calculate
82    /// the byte index based on the index of the character.
83    fn byte_index(&self) -> usize {
84        self.input
85            .char_indices()
86            .map(|(i, _)| i)
87            .nth(self.character_index)
88            .unwrap_or(self.input.len())
89    }
90
91    fn delete_char(&mut self) {
92        let is_not_cursor_leftmost = self.character_index != 0;
93        if is_not_cursor_leftmost {
94            // Method "remove" is not used on the saved text for deleting the selected char.
95            // Reason: Using remove on String works on bytes instead of the chars.
96            // Using remove would require special care because of char boundaries.
97
98            let current_index = self.character_index;
99            let from_left_to_current_index = current_index - 1;
100
101            // Getting all characters before the selected character.
102            let before_char_to_delete = self.input.chars().take(from_left_to_current_index);
103            // Getting all characters after selected character.
104            let after_char_to_delete = self.input.chars().skip(current_index);
105
106            // Put all characters together except the selected one.
107            // By leaving the selected one out, it is forgotten and therefore deleted.
108            self.input = before_char_to_delete.chain(after_char_to_delete).collect();
109            self.move_cursor_left();
110        }
111    }
112
113    fn clamp_cursor(&self, new_cursor_pos: usize) -> usize {
114        new_cursor_pos.clamp(0, self.input.chars().count())
115    }
116
117    const fn reset_cursor(&mut self) {
118        self.character_index = 0;
119    }
120
121    fn submit_message(&mut self) {
122        self.messages.push(self.input.clone());
123        self.input.clear();
124        self.reset_cursor();
125    }
126
127    fn run(mut self, terminal: &mut DefaultTerminal) -> Result<()> {
128        loop {
129            terminal.draw(|frame| self.render(frame))?;
130
131            if let Some(key) = event::read()?.as_key_press_event() {
132                match self.input_mode {
133                    InputMode::Normal => match key.code {
134                        KeyCode::Char('e') => {
135                            self.input_mode = InputMode::Editing;
136                        }
137                        KeyCode::Char('q') => {
138                            return Ok(());
139                        }
140                        _ => {}
141                    },
142                    InputMode::Editing if key.kind == KeyEventKind::Press => match key.code {
143                        KeyCode::Enter => self.submit_message(),
144                        KeyCode::Char(to_insert) => self.enter_char(to_insert),
145                        KeyCode::Backspace => self.delete_char(),
146                        KeyCode::Left => self.move_cursor_left(),
147                        KeyCode::Right => self.move_cursor_right(),
148                        KeyCode::Esc => self.input_mode = InputMode::Normal,
149                        _ => {}
150                    },
151                    InputMode::Editing => {}
152                }
153            }
154        }
155    }
156
157    fn render(&self, frame: &mut Frame) {
158        let layout = Layout::vertical([
159            Constraint::Length(1),
160            Constraint::Length(3),
161            Constraint::Min(1),
162        ]);
163        let [help_area, input_area, messages_area] = frame.area().layout(&layout);
164
165        let (msg, style) = match self.input_mode {
166            InputMode::Normal => (
167                vec![
168                    "Press ".into(),
169                    "q".bold(),
170                    " to exit, ".into(),
171                    "e".bold(),
172                    " to start editing.".bold(),
173                ],
174                Style::default().add_modifier(Modifier::RAPID_BLINK),
175            ),
176            InputMode::Editing => (
177                vec![
178                    "Press ".into(),
179                    "Esc".bold(),
180                    " to stop editing, ".into(),
181                    "Enter".bold(),
182                    " to record the message".into(),
183                ],
184                Style::default(),
185            ),
186        };
187        let text = Text::from(Line::from(msg)).patch_style(style);
188        let help_message = Paragraph::new(text);
189        frame.render_widget(help_message, help_area);
190
191        let input = Paragraph::new(self.input.as_str())
192            .style(match self.input_mode {
193                InputMode::Normal => Style::default(),
194                InputMode::Editing => Style::default().fg(Color::Yellow),
195            })
196            .block(Block::bordered().title("Input"));
197        frame.render_widget(input, input_area);
198        match self.input_mode {
199            // Hide the cursor. `Frame` does this by default, so we don't need to do anything here
200            InputMode::Normal => {}
201
202            // Make the cursor visible and ask ratatui to put it at the specified coordinates after
203            // rendering
204            #[expect(clippy::cast_possible_truncation)]
205            InputMode::Editing => frame.set_cursor_position(Position::new(
206                // Draw the cursor at the current position in the input field.
207                // This position can be controlled via the left and right arrow key
208                input_area.x + self.character_index as u16 + 1,
209                // Move one line down, from the border to the input line
210                input_area.y + 1,
211            )),
212        }
213
214        let messages: Vec<ListItem> = self
215            .messages
216            .iter()
217            .enumerate()
218            .map(|(i, m)| {
219                let content = Line::from(Span::raw(format!("{i}: {m}")));
220                ListItem::new(content)
221            })
222            .collect();
223        let messages = List::new(messages).block(Block::bordered().title("Messages"));
224        frame.render_widget(messages, messages_area);
225    }
226}
