1//! A Ratatui example that demonstrates how to render weather data using [`BarChart`] widget.
2//!
3//! Generates random temperature data for each hour of the day and renders it as a vertical bar.
4//!
5//! This example runs with the Ratatui library code in the branch that you are currently reading.
6//! See the [`latest`] branch for the code which works with the most recent Ratatui release.
7//!
8//! [`latest`]: https://github.com/ratatui/ratatui/tree/latest
9//! [`BarChart`]: https://docs.rs/ratatui/latest/ratatui/widgets/struct.BarChart.html
10
11use color_eyre::Result;
12use crossterm::event;
13use rand::{Rng, rng};
14use ratatui::Frame;
15use ratatui::layout::{Constraint, Layout};
16use ratatui::style::{Color, Style, Stylize};
17use ratatui::text::Line;
18use ratatui::widgets::{Bar, BarChart, BarGroup};
19
20fn main() -> Result<()> {
21    color_eyre::install()?;
22
23    let temperatures: Vec<u8> = (0..24).map(|_| rng().random_range(50..90)).collect();
24    ratatui::run(|terminal| {
25        loop {
26            terminal.draw(|frame| render(frame, &temperatures))?;
27            if event::read()?.is_key_press() {
28                break Ok(());
29            }
30        }
31    })
32}
33
34fn render(frame: &mut Frame, temperatures: &[u8]) {
35    let layout = Layout::vertical([Constraint::Length(1), Constraint::Fill(1)]).spacing(1);
36    let [title, main] = frame.area().layout(&layout);
37
38    frame.render_widget("Weather demo".bold().into_centered_line(), title);
39    frame.render_widget(vertical_barchart(temperatures), main);
40}
41
42/// Create a vertical bar chart from the temperatures data.
43fn vertical_barchart(temperatures: &[u8]) -> BarChart<'_> {
44    let bars: Vec<Bar> = temperatures
45        .iter()
46        .enumerate()
47        .map(|(hour, value)| vertical_bar(hour, value))
48        .collect();
49    BarChart::default()
50        .data(BarGroup::default().bars(&bars))
51        .bar_width(5)
52}
53
54fn vertical_bar(hour: usize, temperature: &u8) -> Bar<'_> {
55    Bar::default()
56        .value(u64::from(*temperature))
57        .label(Line::from(format!("{hour:>02}:00")))
58        .text_value(format!("{temperature:>3}°"))
59        .style(temperature_style(*temperature))
60        .value_style(temperature_style(*temperature).reversed())
61}
62
63/// create a yellow to red value based on the value (50-90)
64fn temperature_style(value: u8) -> Style {
65    let green = (255.0 * (1.0 - f64::from(value - 50) / 40.0)) as u8;
66    let color = Color::Rgb(255, green, 0);
67    Style::new().fg(color)
68}
