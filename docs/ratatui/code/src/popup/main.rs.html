1//! A Ratatui example that demonstrates how to handle popups.
2//! See also:
3//! - <https://github.com/joshka/tui-popup> and
4//! - <https://github.com/sephiroth74/tui-confirm-dialog>
5//!
6//! This example runs with the Ratatui library code in the branch that you are currently
7//! reading. See the [`latest`] branch for the code which works with the most recent Ratatui
8//! release.
9//!
10//! [`latest`]: https://github.com/ratatui/ratatui/tree/latest
11use color_eyre::Result;
12use crossterm::event::{self, KeyCode};
13use ratatui::Frame;
14use ratatui::layout::{Constraint, Flex, Layout, Rect};
15use ratatui::style::Stylize;
16use ratatui::text::Line;
17use ratatui::widgets::{Block, Clear};
18
19fn main() -> Result<()> {
20    color_eyre::install()?;
21
22    // This flag will be toggled when the user presses 'p'. This could be stored in an app struct
23    // if you have more state to manage than just this flag.
24    let mut show_popup = false;
25
26    ratatui::run(|terminal| {
27        loop {
28            terminal.draw(|frame| render(frame, show_popup))?;
29
30            if let Some(key) = event::read()?.as_key_press_event() {
31                match key.code {
32                    KeyCode::Char('q') => return Ok(()),
33                    KeyCode::Char('p') => show_popup = !show_popup,
34                    _ => {}
35                }
36            }
37        }
38    })
39}
40
41fn render(frame: &mut Frame, show_popup: bool) {
42    let area = frame.area();
43
44    let layout = Layout::vertical([Constraint::Length(1), Constraint::Fill(1)]);
45    let [instructions, content] = area.layout(&layout);
46
47    frame.render_widget(
48        Line::from("Press 'p' to toggle popup, 'q' to quit").centered(),
49        instructions,
50    );
51
52    frame.render_widget(Block::bordered().title("Content").on_blue(), content);
53
54    if show_popup {
55        let popup = Block::bordered().title("Popup");
56        let popup_area = centered_area(area, 60, 20);
57        // clears out any background in the area before rendering the popup
58        frame.render_widget(Clear, popup_area);
59        frame.render_widget(popup, popup_area);
60    }
61}
62
63/// Create a centered rect using up certain percentage of the available rect
64fn centered_area(area: Rect, percent_x: u16, percent_y: u16) -> Rect {
65    let vertical = Layout::vertical([Constraint::Percentage(percent_y)]).flex(Flex::Center);
66    let horizontal = Layout::horizontal([Constraint::Percentage(percent_x)]).flex(Flex::Center);
67    let [area] = area.layout(&vertical);
68    let [area] = area.layout(&horizontal);
69    area
