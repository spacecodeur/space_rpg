1/// A Ratatui example that demonstrates how to handle charts.
2///
3/// This example demonstrates how to draw various types of charts such as line, bar, and
4/// scatter charts.
5///
6/// This example runs with the Ratatui library code in the branch that you are currently
7/// reading. See the [`latest`] branch for the code which works with the most recent Ratatui
8/// release.
9///
10/// [`latest`]: https://github.com/ratatui/ratatui/tree/latest
11use std::time::{Duration, Instant};
12
13use color_eyre::Result;
14use crossterm::event::{self, KeyCode};
15use ratatui::layout::{Constraint, Layout, Rect};
16use ratatui::style::{Color, Modifier, Style, Stylize};
17use ratatui::symbols::{self, Marker};
18use ratatui::text::{Line, Span};
19use ratatui::widgets::{Axis, Block, Chart, Dataset, GraphType, LegendPosition};
20use ratatui::{DefaultTerminal, Frame};
21
22fn main() -> Result<()> {
23    color_eyre::install()?;
24    ratatui::run(|terminal| App::new().run(terminal))
25}

27struct App {
28    signal1: SinSignal,
29    data1: Vec<(f64, f64)>,
30    signal2: SinSignal,
31    data2: Vec<(f64, f64)>,
32    window: [f64; 2],
33}
34
35#[derive(Clone)]
36struct SinSignal {
37    x: f64,
38    interval: f64,
39    period: f64,
40    scale: f64,
41}
42
43impl SinSignal {
44    const fn new(interval: f64, period: f64, scale: f64) -> Self {
45        Self {
46            x: 0.0,
47            interval,
48            period,
49            scale,
50        }
51    }
52}
53
54impl Iterator for SinSignal {
55    type Item = (f64, f64);
56    fn next(&mut self) -> Option<Self::Item> {
57        let point = (self.x, (self.x * 1.0 / self.period).sin() * self.scale);
58        self.x += self.interval;
59        Some(point)
60    }
61}
62
63impl App {
64    fn new() -> Self {
65        let mut signal1 = SinSignal::new(0.2, 3.0, 18.0);
66        let mut signal2 = SinSignal::new(0.1, 2.0, 10.0);
67        let data1 = signal1.by_ref().take(200).collect::<Vec<(f64, f64)>>();
68        let data2 = signal2.by_ref().take(200).collect::<Vec<(f64, f64)>>();
69        Self {
70            signal1,
71            data1,
72            signal2,
73            data2,
74            window: [0.0, 20.0],
75        }
76    }
77
78    fn run(mut self, terminal: &mut DefaultTerminal) -> Result<()> {
79        let tick_rate = Duration::from_millis(250);
80        let mut last_tick = Instant::now();
81        loop {
82            terminal.draw(|frame| self.render(frame))?;
83
84            let timeout = tick_rate.saturating_sub(last_tick.elapsed());
85            if !event::poll(timeout)? {
86                self.on_tick();
87                last_tick = Instant::now();
88                continue;
89            }
90            if event::read()?
91                .as_key_press_event()
92                .is_some_and(|key| key.code == KeyCode::Char('q'))
93            {
94                return Ok(());
95            }
96        }
97    }
98
99    fn on_tick(&mut self) {
100        self.data1.drain(0..5);
101        self.data1.extend(self.signal1.by_ref().take(5));
102
103        self.data2.drain(0..10);
104        self.data2.extend(self.signal2.by_ref().take(10));
105
106        self.window[0] += 1.0;
107        self.window[1] += 1.0;
108    }
109
110    fn render(&self, frame: &mut Frame) {
111        let vertical = Layout::vertical([Constraint::Fill(1); 2]);
112        let [top, bottom] = frame.area().layout(&vertical);
113        let horizontal = Layout::horizontal([Constraint::Fill(1), Constraint::Length(29)]);
114        let [animated_chart, bar_chart] = top.layout(&horizontal);
115        let [line_chart, scatter] = bottom.layout(&Layout::horizontal([Constraint::Fill(1); 2]));
116
117        self.render_animated_chart(frame, animated_chart);
118        render_barchart(frame, bar_chart);
119        render_line_chart(frame, line_chart);
120        render_scatter(frame, scatter);
121    }
122
123    fn render_animated_chart(&self, frame: &mut Frame, area: Rect) {
124        let x_labels = vec![
125            Span::styled(
126                format!("{}", self.window[0]),
127                Style::default().add_modifier(Modifier::BOLD),
128            ),
129            Span::raw(format!("{}", f64::midpoint(self.window[0], self.window[1]))),
130            Span::styled(
131                format!("{}", self.window[1]),
132                Style::default().add_modifier(Modifier::BOLD),
133            ),
134        ];
135        let datasets = vec![
136            Dataset::default()
137                .name("data2")
138                .marker(symbols::Marker::Dot)
139                .style(Style::default().fg(Color::Cyan))
140                .data(&self.data1),
141            Dataset::default()
142                .name("data3")
143                .marker(symbols::Marker::Braille)
144                .style(Style::default().fg(Color::Yellow))
145                .data(&self.data2),
146        ];
147
148        let chart = Chart::new(datasets)
149            .block(Block::bordered())
150            .x_axis(
151                Axis::default()
152                    .title("X Axis")
153                    .style(Style::default().fg(Color::Gray))
154                    .labels(x_labels)
155                    .bounds(self.window),
156            )
157            .y_axis(
158                Axis::default()
159                    .title("Y Axis")
160                    .style(Style::default().fg(Color::Gray))
161                    .labels(["-20".bold(), "0".into(), "20".bold()])
162                    .bounds([-20.0, 20.0]),
163            );
164
165        frame.render_widget(chart, area);
166    }
167}

169fn render_barchart(frame: &mut Frame, bar_chart: Rect) {
170    let dataset = Dataset::default()
171        .marker(symbols::Marker::HalfBlock)
172        .style(Style::new().fg(Color::Blue))
173        .graph_type(GraphType::Bar)
174        // a bell curve
175        .data(&[
176            (0., 0.4),
177            (10., 2.9),
178            (20., 13.5),
179            (30., 41.1),
180            (40., 80.1),
181            (50., 100.0),
182            (60., 80.1),
183            (70., 41.1),
184            (80., 13.5),
185            (90., 2.9),
186            (100., 0.4),
187        ]);
188
189    let chart = Chart::new(vec![dataset])
190        .block(Block::bordered().title_top(Line::from("Bar chart").cyan().bold().centered()))
191        .x_axis(
192            Axis::default()
193                .style(Style::default().gray())
194                .bounds([0.0, 100.0])
195                .labels(["0".bold(), "50".into(), "100.0".bold()]),
196        )
197        .y_axis(
198            Axis::default()
199                .style(Style::default().gray())
200                .bounds([0.0, 100.0])
201                .labels(["0".bold(), "50".into(), "100.0".bold()]),
202        )
203        .hidden_legend_constraints((Constraint::Ratio(1, 2), Constraint::Ratio(1, 2)));
204
205    frame.render_widget(chart, bar_chart);
206}

208fn render_line_chart(frame: &mut Frame, area: Rect) {
209    let datasets = vec![
210        Dataset::default()
211            .name("Line from only 2 points".italic())
212            .marker(symbols::Marker::Braille)
213            .style(Style::default().fg(Color::Yellow))
214            .graph_type(GraphType::Line)
215            .data(&[(1., 1.), (4., 4.)]),
216    ];
217
218    let chart = Chart::new(datasets)
219        .block(Block::bordered().title(Line::from("Line chart").cyan().bold().centered()))
220        .x_axis(
221            Axis::default()
222                .title("X Axis")
223                .style(Style::default().gray())
224                .bounds([0.0, 5.0])
225                .labels(["0".bold(), "2.5".into(), "5.0".bold()]),
226        )
227        .y_axis(
228            Axis::default()
229                .title("Y Axis")
230                .style(Style::default().gray())
231                .bounds([0.0, 5.0])
232                .labels(["0".bold(), "2.5".into(), "5.0".bold()]),
233        )
234        .legend_position(Some(LegendPosition::TopLeft))
235        .hidden_legend_constraints((Constraint::Ratio(1, 2), Constraint::Ratio(1, 2)));
236
237    frame.render_widget(chart, area);
238}

240fn render_scatter(frame: &mut Frame, area: Rect) {
241    let datasets = vec![
242        Dataset::default()
243            .name("Heavy")
244            .marker(Marker::Dot)
245            .graph_type(GraphType::Scatter)
246            .style(Style::new().yellow())
247            .data(&HEAVY_PAYLOAD_DATA),
248        Dataset::default()
249            .name("Medium".underlined())
250            .marker(Marker::Braille)
251            .graph_type(GraphType::Scatter)
252            .style(Style::new().magenta())
253            .data(&MEDIUM_PAYLOAD_DATA),
254        Dataset::default()
255            .name("Small")
256            .marker(Marker::Dot)
257            .graph_type(GraphType::Scatter)
258            .style(Style::new().cyan())
259            .data(&SMALL_PAYLOAD_DATA),
260    ];
261
262    let chart = Chart::new(datasets)
263        .block(Block::bordered().title(Line::from("Scatter chart").cyan().bold().centered()))
264        .x_axis(
265            Axis::default()
266                .title("Year")
267                .bounds([1960., 2020.])
268                .style(Style::default().fg(Color::Gray))
269                .labels(["1960", "1990", "2020"]),
270        )
271        .y_axis(
272            Axis::default()
273                .title("Cost")
274                .bounds([0., 75000.])
275                .style(Style::default().fg(Color::Gray))
276                .labels(["0", "37 500", "75 000"]),
277        )
278        .hidden_legend_constraints((Constraint::Ratio(1, 2), Constraint::Ratio(1, 2)));
279
280    frame.render_widget(chart, area);
281}
