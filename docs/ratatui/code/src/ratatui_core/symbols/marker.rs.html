5/// Marker to use when plotting data points
6#[derive(Debug, Default, Display, EnumString, Clone, Copy, Eq, PartialEq, Hash)]
7pub enum Marker {
8    /// One point per cell in shape of dot (`•`)
9    #[default]
10    Dot,
11    /// One point per cell in shape of a block (`█`)
12    Block,
13    /// One point per cell in the shape of a bar (`▄`)
14    Bar,
15    /// Use the [Unicode Braille Patterns](https://en.wikipedia.org/wiki/Braille_Patterns) block to
16    /// represent data points.
17    ///
18    /// This is a 2x4 grid of dots, where each dot can be either on or off.
19    ///
20    /// Note: Support for this marker is limited to terminals and fonts that support Unicode
21    /// Braille Patterns. If your terminal does not support this, you will see unicode replacement
22    /// characters (`�`) instead of Braille dots (`⠓`, `⣇`, `⣿`).
23    Braille,
24    /// Use the unicode block and half block characters (`█`, `▄`, and `▀`) to represent points in
25    /// a grid that is double the resolution of the terminal. Because each terminal cell is
26    /// generally about twice as tall as it is wide, this allows for a square grid of pixels.
27    HalfBlock,
28}

38    #[test]
39    fn marker_tostring() {

42        assert_eq!(Marker::Bar.to_string(), "Bar");
43        assert_eq!(Marker::Braille.to_string(), "Braille");
44    }
45
46    #[test]
47    fn marker_from_str() {
48        assert_eq!("Dot".parse::<Marker>(), Ok(Marker::Dot));
49        assert_eq!("Block".parse::<Marker>(), Ok(Marker::Block));
50        assert_eq!("Bar".parse::<Marker>(), Ok(Marker::Bar));
51        assert_eq!("Braille".parse::<Marker>(), Ok(Marker::Braille));
52        assert_eq!("".parse::<Marker>(), Err(ParseError::VariantNotFound));
