143/// Tabs for the different layouts
144///
145/// Note: the order of the variants will determine the order of the tabs this uses several derive
146/// macros from the `strum` crate to make it easier to iterate over the variants.
147/// (`FromRepr`,`Display`,`EnumIter`).
148#[derive(Default, Debug, Copy, Clone, PartialEq, Eq, FromRepr, Display, EnumIter)]
149enum SelectedTab {
150    #[default]
151    Legacy,
152    Start,
153    Center,
154    End,
155    SpaceAround,
156    SpaceEvenly,
157    SpaceBetween,
158}

160impl App {
161    fn run(mut self, terminal: &mut DefaultTerminal) -> Result<()> {
162        // increase the layout cache to account for the number of layout events. This ensures that
163        // layout is not generally reprocessed on every frame (which would lead to possible janky
164        // results when there are more than one possible solution to the requested layout). This
165        // assumes the user changes spacing about a 100 times or so.
166        let cache_size = EXAMPLE_DATA.len() * SelectedTab::iter().len() * 100;
167        Layout::init_cache(NonZeroUsize::new(cache_size).unwrap());
168
169        while self.is_running() {
170            terminal.draw(|frame| frame.render_widget(self, frame.area()))?;
171            self.handle_events()?;
172        }
173        Ok(())
174    }
175
176    fn is_running(self) -> bool {
177        self.state == AppState::Running
178    }
179
180    fn handle_events(&mut self) -> Result<()> {
181        if let Some(key) = event::read()?.as_key_press_event() {
182            match key.code {
183                KeyCode::Char('q') | KeyCode::Esc => self.quit(),
184                KeyCode::Char('l') | KeyCode::Right => self.next(),
185                KeyCode::Char('h') | KeyCode::Left => self.previous(),
186                KeyCode::Char('j') | KeyCode::Down => self.down(),
187                KeyCode::Char('k') | KeyCode::Up => self.up(),
188                KeyCode::Char('g') | KeyCode::Home => self.top(),
189                KeyCode::Char('G') | KeyCode::End => self.bottom(),
190                KeyCode::Char('+') => self.increment_spacing(),
191                KeyCode::Char('-') => self.decrement_spacing(),
192                _ => (),
193            }
194        }
195        Ok(())
196    }
197
198    fn next(&mut self) {
199        self.selected_tab = self.selected_tab.next();
200    }
201
202    fn previous(&mut self) {
203        self.selected_tab = self.selected_tab.previous();
204    }
205
206    const fn up(&mut self) {
207        self.scroll_offset = self.scroll_offset.saturating_sub(1);
208    }
209
210    fn down(&mut self) {
211        self.scroll_offset = self
212            .scroll_offset
213            .saturating_add(1)
214            .min(max_scroll_offset());
215    }
216
217    const fn top(&mut self) {
218        self.scroll_offset = 0;
219    }
220
221    fn bottom(&mut self) {
222        self.scroll_offset = max_scroll_offset();
223    }
224
225    const fn increment_spacing(&mut self) {
226        self.spacing = self.spacing.saturating_add(1);
227    }
228
229    const fn decrement_spacing(&mut self) {
230        self.spacing = self.spacing.saturating_sub(1);
231    }
232
233    const fn quit(&mut self) {
234        self.state = AppState::Quit;
235    }
236}

238// when scrolling, make sure we don't scroll past the last example
239fn max_scroll_offset() -> u16 {
240    example_height()
241        - EXAMPLE_DATA
242            .last()
243            .map_or(0, |(desc, _)| get_description_height(desc) + 4)
244}
245
246/// The height of all examples combined
247///
248/// Each may or may not have a title so we need to account for that.
249fn example_height() -> u16 {
250    EXAMPLE_DATA
251        .iter()
252        .map(|(desc, _)| get_description_height(desc) + 4)
253        .sum()
254}

256impl Widget for App {
257    fn render(self, area: Rect, buf: &mut Buffer) {
258        let layout = Layout::vertical([Length(3), Length(1), Fill(0)]);
259        let [tabs, axis, demo] = area.layout(&layout);
260        self.tabs().render(tabs, buf);
261        let scroll_needed = self.render_demo(demo, buf);
262        let axis_width = if scroll_needed {
263            axis.width.saturating_sub(1)
264        } else {
265            axis.width
266        };
267        Self::axis(axis_width, self.spacing).render(axis, buf);
268    }
269}

271impl App {
272    fn tabs(self) -> impl Widget {
273        let tab_titles = SelectedTab::iter().map(SelectedTab::to_tab_title);
274        let block = Block::new()
275            .title("Flex Layouts ".bold())
276            .title(" Use ◄ ► to change tab, ▲ ▼  to scroll, - + to change spacing ");
277        Tabs::new(tab_titles)
278            .block(block)
279            .highlight_style(Modifier::REVERSED)
280            .select(self.selected_tab as usize)
281            .divider(" ")
282            .padding("", "")
283    }
284
285    /// a bar like `<----- 80 px (gap: 2 px)? ----->`
286    fn axis(width: u16, spacing: u16) -> impl Widget {
287        let width = width as usize;
288        // only show gap when spacing is not zero
289        let label = if spacing != 0 {
290            format!("{width} px (gap: {spacing} px)")
291        } else {
292            format!("{width} px")
293        };
294        let bar_width = width.saturating_sub(2); // we want to `<` and `>` at the ends
295        let width_bar = format!("<{label:-^bar_width$}>");
296        Paragraph::new(width_bar.dark_gray()).centered()
297    }
298
299    /// Render the demo content
300    ///
301    /// This function renders the demo content into a separate buffer and then splices the buffer
302    /// into the main buffer. This is done to make it possible to handle scrolling easily.
303    ///
304    /// Returns bool indicating whether scroll was needed
305    #[expect(clippy::cast_possible_truncation)]
306    fn render_demo(self, area: Rect, buf: &mut Buffer) -> bool {
307        // render demo content into a separate buffer so all examples fit we add an extra
308        // area.height to make sure the last example is fully visible even when the scroll offset is
309        // at the max
310        let height = example_height();
311        let demo_area = Rect::new(0, 0, area.width, height);
312        let mut demo_buf = Buffer::empty(demo_area);
313
314        let scrollbar_needed = self.scroll_offset != 0 || height > area.height;
315        let content_area = if scrollbar_needed {
316            Rect {
317                width: demo_area.width - 1,
318                ..demo_area
319            }
320        } else {
321            demo_area
322        };
323
324        let mut spacing = self.spacing;
325        self.selected_tab
326            .render(content_area, &mut demo_buf, &mut spacing);
327
328        let visible_content = demo_buf
329            .content
330            .into_iter()
331            .skip((area.width * self.scroll_offset) as usize)
332            .take(area.area() as usize);
333        for (i, cell) in visible_content.enumerate() {
334            let x = i as u16 % area.width;
335            let y = i as u16 / area.width;
336            buf[(area.x + x, area.y + y)] = cell;
337        }
338
339        if scrollbar_needed {
340            let area = area.intersection(buf.area);
341            let mut state = ScrollbarState::new(max_scroll_offset() as usize)
342                .position(self.scroll_offset as usize);
343            Scrollbar::new(ScrollbarOrientation::VerticalRight).render(area, buf, &mut state);
344        }
345        scrollbar_needed
346    }
347}

349impl SelectedTab {
350    /// Get the previous tab, if there is no previous tab return the current tab.
351    fn previous(self) -> Self {
352        let current_index: usize = self as usize;
353        let previous_index = current_index.saturating_sub(1);
354        Self::from_repr(previous_index).unwrap_or(self)
355    }
356
357    /// Get the next tab, if there is no next tab return the current tab.
358    fn next(self) -> Self {
359        let current_index = self as usize;
360        let next_index = current_index.saturating_add(1);
361        Self::from_repr(next_index).unwrap_or(self)
362    }
363
364    /// Convert a `SelectedTab` into a `Line` to display it by the `Tabs` widget.
365    fn to_tab_title(value: Self) -> Line<'static> {
366        use tailwind::{INDIGO, ORANGE, SKY};
367        let text = value.to_string();
368        let color = match value {
369            Self::Legacy => ORANGE.c400,
370            Self::Start => SKY.c400,
371            Self::Center => SKY.c300,
372            Self::End => SKY.c200,
373            Self::SpaceEvenly => INDIGO.c400,
374            Self::SpaceBetween => INDIGO.c300,
375            Self::SpaceAround => INDIGO.c500,
376        };
377        format!(" {text} ").fg(color).bg(Color::Black).into()
378    }
379}

381impl StatefulWidget for SelectedTab {
382    type State = u16;
383    fn render(self, area: Rect, buf: &mut Buffer, spacing: &mut Self::State) {
384        let spacing = *spacing;
385        match self {
386            Self::Legacy => Self::render_examples(area, buf, Flex::Legacy, spacing),
387            Self::Start => Self::render_examples(area, buf, Flex::Start, spacing),
388            Self::Center => Self::render_examples(area, buf, Flex::Center, spacing),
389            Self::End => Self::render_examples(area, buf, Flex::End, spacing),
390            Self::SpaceEvenly => Self::render_examples(area, buf, Flex::SpaceEvenly, spacing),
391            Self::SpaceBetween => Self::render_examples(area, buf, Flex::SpaceBetween, spacing),
392            Self::SpaceAround => Self::render_examples(area, buf, Flex::SpaceAround, spacing),
393        }
394    }
395}

397impl SelectedTab {
398    fn render_examples(area: Rect, buf: &mut Buffer, flex: Flex, spacing: u16) {
399        let heights = EXAMPLE_DATA
400            .iter()
401            .map(|(desc, _)| get_description_height(desc) + 4);
402        let areas = Layout::vertical(heights).flex(Flex::Start).split(area);
403        for (area, (description, constraints)) in areas.iter().zip(EXAMPLE_DATA.iter()) {
404            Example::new(constraints, description, flex, spacing).render(*area, buf);
405        }
406    }
407}

409impl Example {
410    fn new(constraints: &[Constraint], description: &str, flex: Flex, spacing: u16) -> Self {
411        Self {
412            constraints: constraints.into(),
413            description: description.into(),
414            flex,
415            spacing,
416        }
417    }

420impl Widget for Example {
421    fn render(self, area: Rect, buf: &mut Buffer) {
422        let title_height = get_description_height(&self.description);
423        let layout = Layout::vertical([Length(title_height), Fill(0)]);
424        let [title, illustrations] = area.layout(&layout);
425
426        let (blocks, spacers) = Layout::horizontal(&self.constraints)
427            .flex(self.flex)
428            .spacing(self.spacing)
429            .split_with_spacers(illustrations);
430
431        if !self.description.is_empty() {
432            Paragraph::new(
433                self.description
434                    .split('\n')
435                    .map(|s| format!("// {s}").italic().fg(tailwind::SLATE.c400))
436                    .map(Line::from)
437                    .collect::<Vec<Line>>(),
438            )
439            .render(title, buf);
440        }
441
442        for (block, constraint) in blocks.iter().zip(&self.constraints) {
443            Self::illustration(*constraint, block.width).render(*block, buf);
444        }
445
446        for spacer in spacers.iter() {
447            Self::render_spacer(*spacer, buf);
448        }
449    }
450}

452impl Example {
453    fn render_spacer(spacer: Rect, buf: &mut Buffer) {
454        if spacer.width > 1 {
455            let corners_only = symbols::border::Set {
456                top_left: line::NORMAL.top_left,
457                top_right: line::NORMAL.top_right,
458                bottom_left: line::NORMAL.bottom_left,
459                bottom_right: line::NORMAL.bottom_right,
460                vertical_left: " ",
461                vertical_right: " ",
462                horizontal_top: " ",
463                horizontal_bottom: " ",
464            };
465            Block::bordered()
466                .border_set(corners_only)
467                .border_style(Style::reset().dark_gray())
468                .render(spacer, buf);
469        } else {
470            Paragraph::new(Text::from(vec![
471                Line::from(""),
472                Line::from("│"),
473                Line::from("│"),
474                Line::from(""),
475            ]))
476            .style(Style::reset().dark_gray())
477            .render(spacer, buf);
478        }
479        let width = spacer.width;
480        let label = if width > 4 {
481            format!("{width} px")
482        } else if width > 2 {
483            format!("{width}")
484        } else {
485            String::new()
486        };
487        let text = Text::from(vec![
488            Line::raw(""),
489            Line::raw(""),
490            Line::styled(label, Style::reset().dark_gray()),
491        ]);
492        Paragraph::new(text)
493            .style(Style::reset().dark_gray())
494            .alignment(Alignment::Center)
495            .render(spacer, buf);
496    }
497
498    fn illustration(constraint: Constraint, width: u16) -> impl Widget {
499        let main_color = color_for_constraint(constraint);
500        let fg_color = Color::White;
501        let title = format!("{constraint}");
502        let content = format!("{width} px");
503        let text = format!("{title}\n{content}");
504        let block = Block::bordered()
505            .border_set(symbols::border::QUADRANT_OUTSIDE)
506            .border_style(Style::reset().fg(main_color).reversed())
507            .style(Style::default().fg(fg_color).bg(main_color));
508        Paragraph::new(text).centered().block(block)
509    }
510}
511
512const fn color_for_constraint(constraint: Constraint) -> Color {
513    use tailwind::{BLUE, SLATE};
514    match constraint {
515        Constraint::Min(_) => BLUE.c900,
516        Constraint::Max(_) => BLUE.c800,
517        Constraint::Length(_) => SLATE.c700,
518        Constraint::Percentage(_) => SLATE.c800,
519        Constraint::Ratio(_, _) => SLATE.c900,
520        Constraint::Fill(_) => SLATE.c950,
521    }
522}

524#[expect(clippy::cast_possible_truncation)]
525fn get_description_height(s: &str) -> u16 {
526    if s.is_empty() {
527        0
528    } else {
529        s.split('\n').count() as u16
530    }
