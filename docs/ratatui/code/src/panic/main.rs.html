1/// A Ratatui example that demonstrates how to handle panics in your application.
2///
3/// Prior to Ratatui 0.28.1, a panic hook had to be manually set up to ensure that the terminal
4/// was reset when a panic occurred. This was necessary because a panic would interrupt the
5/// normal control flow and leave the terminal in a distorted state.
6///
7/// Starting with Ratatui 0.28.1, the panic hook is automatically set up by the new
8/// `ratatui::init` function, so you no longer need to manually set up the panic hook. This
9/// example now demonstrates how the panic hook acts when it is enabled by default.
10///
11/// When exiting normally or when handling `Result::Err`, we can reset the terminal manually at
12/// the end of `main` just before we print the error.
13///
14/// Because a panic interrupts the normal control flow, manually resetting the terminal at the
15/// end of `main` won't do us any good. Instead, we need to make sure to set up a panic hook
16/// that first resets the terminal before handling the panic. This both reuses the standard
17/// panic hook to ensure a consistent panic handling UX and properly resets the terminal to not
18/// distort the output.
19///
20/// That's why this example is set up to show both situations, with and without the panic hook,
21/// to see the difference.
22///
23/// For more information on how to set this up manually, see the [Color Eyre recipe] in the
24/// Ratatui website.
25///
26/// This example runs with the Ratatui library code in the branch that you are currently
27/// reading. See the [`latest`] branch for the code which works with the most recent Ratatui
28/// release.
29///
30/// [`latest`]: https://github.com/ratatui/ratatui/tree/latest
31/// [Color Eyre recipe]: https://ratatui.rs/recipes/apps/color-eyre
32use color_eyre::{Result, eyre::bail};
33use crossterm::event::{self, KeyCode};
34use ratatui::Frame;
35use ratatui::text::Line;
36use ratatui::widgets::{Block, Paragraph};
37
38#[derive(Debug)]
39enum PanicHandlerState {
40    Enabled,
41    Disabled,
42}
43
44fn main() -> Result<()> {
45    color_eyre::install()?;
46
47    let mut panic_hook_state = PanicHandlerState::Enabled;
48    ratatui::run(|terminal| {
49        loop {
50            terminal.draw(|frame| render(frame, &panic_hook_state))?;
51            if let Some(key) = event::read()?.as_key_press_event() {
52                match key.code {
53                    KeyCode::Char('p') => panic!("intentional demo panic"),
54                    KeyCode::Char('e') => bail!("intentional demo error"),
55                    KeyCode::Char('h') => {
56                        let _ = std::panic::take_hook();
57                        panic_hook_state = PanicHandlerState::Disabled;
58                    }
59                    KeyCode::Char('q') => return Ok(()),
60                    _ => {}
61                }
62            }
63        }
64    })
65}
66
67fn render(frame: &mut Frame, state: &PanicHandlerState) {
68    let text = vec![
69        Line::from(format!("Panic hook is currently: {state:?}")),
70        Line::from(""),
71        Line::from("Press `p` to cause a panic"),
72        Line::from("Press `e` to cause an error"),
73        Line::from("Press `h` to disable the panic hook"),
74        Line::from("Press `q` to quit"),
75        Line::from(""),
76        Line::from("When your app panics without a panic hook, you will likely have to"),
77        Line::from("reset your terminal afterwards with the `reset` command"),
78        Line::from(""),
79        Line::from("Try first with the panic handler enabled, and then with it disabled"),
80        Line::from("to see the difference"),
81    ];
82
83    let paragraph = Paragraph::new(text)
84        .block(Block::bordered().title("Panic Handler Demo"))
85        .centered();
86
87    frame.render_widget(paragraph, frame.area());
88}
