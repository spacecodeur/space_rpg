40fn main() -> Result<()> {
41    color_eyre::install()?;
42    // this size is to match the size of the terminal when running the demo
43    // using vhs in a 1280x640 sized window (github social preview size)
44    let viewport = Viewport::Fixed(Rect::new(0, 0, 81, 18));
45    let terminal = ratatui::init_with_options(TerminalOptions { viewport });
46    execute!(stdout(), EnterAlternateScreen).expect("failed to enter alternate screen");
47    let app_result = App::default().run(terminal);
48    execute!(stdout(), LeaveAlternateScreen).expect("failed to leave alternate screen");
49    ratatui::restore();
50    app_result
51}
