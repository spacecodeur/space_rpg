13#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
14pub struct WeatherTab {
15    pub download_progress: usize,
16}
17
18impl WeatherTab {
19    /// Simulate a download indicator by decrementing the row index.
20    pub fn prev(&mut self) {
21        self.download_progress = self.download_progress.saturating_sub(1);
22    }
23
24    /// Simulate a download indicator by incrementing the row index.
25    pub fn next(&mut self) {
26        self.download_progress = self.download_progress.saturating_add(1);
27    }
28}

30impl Widget for WeatherTab {
31    fn render(self, area: Rect, buf: &mut Buffer) {
32        RgbSwatch.render(area, buf);
33        let area = area.inner(Margin {
34            vertical: 1,
35            horizontal: 2,
36        });
37        Clear.render(area, buf);
38        Block::new().style(THEME.content).render(area, buf);
39
40        let area = area.inner(Margin {
41            horizontal: 2,
42            vertical: 1,
43        });
44        let tab_layout = Layout::vertical([
45            Constraint::Min(0),
46            Constraint::Length(1),
47            Constraint::Length(1),
48        ]);
49        let [main, _, gauges] = area.layout(&tab_layout);
50        let main_layout = Layout::horizontal([Constraint::Length(23), Constraint::Min(0)]);
51        let [calendar, charts] = main.layout(&main_layout);
52        let charts_layout = Layout::vertical([Constraint::Length(29), Constraint::Min(0)]);
53        let [simple, horizontal] = charts.layout(&charts_layout);
54
55        render_calendar(calendar, buf);
56        render_simple_barchart(simple, buf);
57        render_horizontal_barchart(horizontal, buf);
58        render_gauge(self.download_progress, gauges, buf);
59    }
60}

62fn render_calendar(area: Rect, buf: &mut Buffer) {
63    let date = OffsetDateTime::now_utc().date();
64    Monthly::new(date, CalendarEventStore::today(Style::new().red().bold()))
65        .block(Block::new().padding(Padding::new(0, 0, 2, 0)))
66        .show_month_header(Style::new().bold())
67        .show_weekdays_header(Style::new().italic())
68        .render(area, buf);
69}

71fn render_simple_barchart(area: Rect, buf: &mut Buffer) {
72    let data = [
73        ("Sat", 76),
74        ("Sun", 69),
75        ("Mon", 65),
76        ("Tue", 67),
77        ("Wed", 65),
78        ("Thu", 69),
79        ("Fri", 73),
80    ];
81    let data = data
82        .into_iter()
83        .map(|(label, value)| {
84            Bar::default()
85                .value(value)
86                // This doesn't actually render correctly as the text is too wide for the bar
87                // See https://github.com/ratatui/ratatui/issues/513 for more info
88                // (the demo GIFs hack around this by hacking the calculation in bars.rs)
89                .text_value(format!("{value}°"))
90                .style(if value > 70 {
91                    Style::new().fg(Color::Red)
92                } else {
93                    Style::new().fg(Color::Yellow)
94                })
95                .value_style(if value > 70 {
96                    Style::new().fg(Color::Gray).bg(Color::Red).bold()
97                } else {
98                    Style::new().fg(Color::DarkGray).bg(Color::Yellow).bold()
99                })
100                .label(label)
101        })
102        .collect_vec();
103    let group = BarGroup::default().bars(&data);
104    BarChart::default()
105        .data(group)
106        .bar_width(3)
107        .bar_gap(1)
108        .render(area, buf);
109}

111fn render_horizontal_barchart(area: Rect, buf: &mut Buffer) {

133#[expect(clippy::cast_precision_loss)]
134pub fn render_gauge(progress: usize, area: Rect, buf: &mut Buffer) {
135    let percent = (progress * 3).min(100) as f64;
136
137    render_line_gauge(percent, area, buf);
138}

140#[expect(clippy::cast_possible_truncation)]
141fn render_line_gauge(percent: f64, area: Rect, buf: &mut Buffer) {
142    // cycle color hue based on the percent for a neat effect yellow -> red
143    let hue = 90.0 - (percent as f32 * 0.6);
144    let value = Okhsv::max_value();
145    let filled_color = color_from_oklab(hue, Okhsv::max_saturation(), value);
146    let unfilled_color = color_from_oklab(hue, Okhsv::max_saturation(), value * 0.5);
147    let label = if percent < 100.0 {
148        format!("Downloading: {percent}%")
149    } else {
150        "Download Complete!".into()
151    };
152    LineGauge::default()
153        .ratio(percent / 100.0)
154        .label(label)
155        .style(Style::new().light_blue())
156        .filled_style(Style::new().fg(filled_color))
157        .unfilled_style(Style::new().fg(unfilled_color))
158        .filled_symbol(symbols::line::THICK_HORIZONTAL)
159        .unfilled_symbol(symbols::line::THICK_HORIZONTAL)
160        .render(area, buf);
161}
