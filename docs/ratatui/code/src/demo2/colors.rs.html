7/// A widget that renders a color swatch of RGB colors.
8///
9/// The widget is rendered as a rectangle with the hue changing along the x-axis from 0.0 to 360.0
10/// and the value changing along the y-axis (from 1.0 to 0.0). Each pixel is rendered as a block
11/// character with the top half slightly lighter than the bottom half.
12pub struct RgbSwatch;
13
14impl Widget for RgbSwatch {
15    #[expect(clippy::cast_precision_loss, clippy::similar_names)]
16    fn render(self, area: Rect, buf: &mut Buffer) {
17        for (yi, y) in (area.top()..area.bottom()).enumerate() {
18            let value = f32::from(area.height) - yi as f32;
19            let value_fg = value / f32::from(area.height);
20            let value_bg = (value - 0.5) / f32::from(area.height);
21            for (xi, x) in (area.left()..area.right()).enumerate() {
22                let hue = xi as f32 * 360.0 / f32::from(area.width);
23                let fg = color_from_oklab(hue, Okhsv::max_saturation(), value_fg);
24                let bg = color_from_oklab(hue, Okhsv::max_saturation(), value_bg);
25                buf[(x, y)].set_char('▀').set_fg(fg).set_bg(bg);
26            }
27        }
28    }
29}
30
31/// Convert a hue and value into an RGB color via the Oklab color space.
32///
33/// See <https://bottosson.github.io/posts/oklab/> for more details.
34pub fn color_from_oklab(hue: f32, saturation: f32, value: f32) -> Color {
35    let color: Srgb = Okhsv::new(hue, saturation, value).into_color();
36    let color = color.into_format();
37    Color::Rgb(color.red, color.green, color.blue)
38}
